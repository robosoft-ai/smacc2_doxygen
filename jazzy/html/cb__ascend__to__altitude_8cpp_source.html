41std::vector<NedPoint> CbAscendToAltitude::buildPath(const NedPoint & current)
42{
43 // vertical move: speed = climb rate, keep heading, no prepend (path already
44 // starts at the current position), leash sized for the climb rate
45 followerParams_.groundSpeed = std::max(params_.climbRate, 0.1f);
46 followerParams_.leash = std::max(followerParams_.leash, 2.0f * followerParams_.groundSpeed);
47 followerParams_.yawMode = YawMode::HOLD_ENTRY;
48 followerParams_.prependCurrentPosition = false;
49 followerParams_.arrivalZTol = std::min(followerParams_.arrivalZTol, params_.tolerance);
50
51 RCLCPP_INFO(
52 getLogger(), "CbAscendToAltitude: %.1f m AGL -> %.1f m AGL at %.1f m/s", -current.z,
53 params_.altitudeAgl, followerParams_.groundSpeed);
54 return generateFlightPatternAscend(params_, current);
55}
