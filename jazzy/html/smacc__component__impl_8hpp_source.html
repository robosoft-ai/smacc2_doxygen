42template <typename TComponent>
43void ISmaccComponent::requiresComponent(
44 TComponent *& requiredComponentStorage, ComponentRequirement requirementType)
45{
46 requiredComponentStorage = this->owner_->getComponent<TComponent>();
47
48 if (requiredComponentStorage == nullptr && requirementType == ComponentRequirement::HARD)
49 {
50 RCLCPP_DEBUG_STREAM(
51 this->getLogger(), std::string("Required component ") +
52 demangleSymbol(typeid(TComponent).name()) +
53 " not found. Available components:");
54
55 std::vector<std::shared_ptr<ISmaccComponent>> components;
56 this->owner_->getComponents(components);
57
58 for (auto c : components)
59 {
60 RCLCPP_DEBUG(this->getLogger(), "- Component %s", c->getName().c_str());
61 }
62
63 throw std::runtime_error(
64 std::string("Component ") + demangleSymbol(typeid(TComponent).name()) + " not found");
65 }
66}
