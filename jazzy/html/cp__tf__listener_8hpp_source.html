59 inline void onInitialize() override
60 {
61 // Take strong references to the process-wide buffer and listener, creating them if no
62 // other instance holds them. This instance then owns them for its own lifetime; the statics
63 // below are weak, so the pair is destroyed with the last CpTfListener, inside the state
64 // machine's destruction and before the rclcpp context is finalized. A static shared_ptr used
65 // to hold the listener until static destruction after main() returned, when its node's
66 // teardown ran against a finalized DDS participant and crashed (finding R, cl_nav2z sibling).
67 std::lock_guard<std::mutex> guard(tfMutex_);
68 tfBuffer_ = sharedTfBuffer_.lock();
69 tfListener_ = sharedTfListener_.lock();
70 if (!tfBuffer_ || !tfListener_)
71 {
72 RCLCPP_INFO(getLogger(), "[CpTfListener] Initializing shared TF2 buffer and listener");
73 tfBuffer_ = std::make_shared<tf2_ros::Buffer>(getNode()->get_clock());
74 tfListener_ = std::make_shared<tf2_ros::TransformListener>(*tfBuffer_);
75 sharedTfBuffer_ = tfBuffer_;
76 sharedTfListener_ = tfListener_;
77 }
78 else
79 {
80 RCLCPP_DEBUG(getLogger(), "[CpTfListener] Reusing existing shared TF2 resources");
81 }
82 }

92 inline std::optional<geometry_msgs::msg::TransformStamped> lookupTransform(
93 const std::string & target_frame, const std::string & source_frame,
94 const rclcpp::Time & time = rclcpp::Time(0))
95 {
96 std::lock_guard<std::mutex> guard(tfMutex_);
97
98 if (!tfBuffer_)
99 {
100 RCLCPP_ERROR(
101 getLogger(), "[CpTfListener] TF buffer not initialized. Call onInitialize() first.");
102 return std::nullopt;
103 }
104
105 try
106 {
107 auto transform = tfBuffer_->lookupTransform(target_frame, source_frame, time);
108 return transform;
109 }
110 catch (const tf2::TransformException & ex)
111 {
112 RCLCPP_WARN(
113 getLogger(), "[CpTfListener] Could not get transform from '%s' to '%s': %s",
114 source_frame.c_str(), target_frame.c_str(), ex.what());
115 return std::nullopt;
116 }
117 }

126 inline std::optional<geometry_msgs::msg::PoseStamped> transformPose(
127 const geometry_msgs::msg::PoseStamped & pose, const std::string & target_frame)
128 {
129 std::lock_guard<std::mutex> guard(tfMutex_);
130
131 if (!tfBuffer_)
132 {
133 RCLCPP_ERROR(
134 getLogger(), "[CpTfListener] TF buffer not initialized. Call onInitialize() first.");
135 return std::nullopt;
136 }
137
138 try
139 {
140 geometry_msgs::msg::PoseStamped transformed_pose;
141 tfBuffer_->transform(pose, transformed_pose, target_frame);
142 return transformed_pose;
143 }
144 catch (const tf2::TransformException & ex)
145 {
146 RCLCPP_WARN(
147 getLogger(), "[CpTfListener] Could not transform pose to frame '%s': %s",
148 target_frame.c_str(), ex.what());
149 return std::nullopt;
150 }
151 }

162 inline bool canTransform(

186 inline std::shared_ptr<tf2_ros::Buffer> getBuffer()
187 {
188 std::lock_guard<std::mutex> guard(tfMutex_);
189 return tfBuffer_;
190 }
191
192private:
193 // Per-instance strong references. Buffer before listener: the listener references the buffer
194 // and must be destroyed first.
195 std::shared_ptr<tf2_ros::Buffer> tfBuffer_;
196 std::shared_ptr<tf2_ros::TransformListener> tfListener_;
197
198 // Process-wide weak references so every instance shares one buffer and listener while any
199 // instance is alive, without extending their lifetime past the last instance. `inline static`
200 // so a header-only component defines each once per program, not once per translation unit.
201 inline static std::weak_ptr<tf2_ros::Buffer> sharedTfBuffer_;
202 inline static std::weak_ptr<tf2_ros::TransformListener> sharedTfListener_;
203 inline static std::mutex tfMutex_;
204};
