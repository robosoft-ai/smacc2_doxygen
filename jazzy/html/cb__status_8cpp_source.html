20void CbStatus::onEntry()
21{
22 this->requiresClient(client_);
23
24 if (!client_)
25 {
26 RCLCPP_ERROR(getLogger(), "[CbStatus] ClGcalcli client not available");
27 connection_state_ = ConnectionState::ERROR;
28 return;
29 }
30
31 auto * connection = client_->getConnection();
32 auto * poller = client_->getPoller();
33
34 if (connection)
35 {
36 connection_state_ = connection->getConnectionState();
37 }
38 else
39 {
40 connection_state_ = ConnectionState::DISCONNECTED;
41 }
42
43 if (poller)
44 {
45 events_ = poller->getEvents();
46 active_events_ = poller->getActiveEvents();
47 }
48
49 RCLCPP_INFO(
50 getLogger(), "[CbStatus] Connection: %s, Events: %zu, Active: %zu",
51 connection_state_ == ConnectionState::CONNECTED ? "CONNECTED"
52 : connection_state_ == ConnectionState::DISCONNECTED ? "DISCONNECTED"
53 : connection_state_ == ConnectionState::AUTHENTICATING ? "AUTHENTICATING"
54 : "ERROR",
55 events_.size(), active_events_.size());
56}
