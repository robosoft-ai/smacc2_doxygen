25void CbLand::onEntry()
26{
27 this->requiresComponent(vehicleCommand_);
28 this->requiresComponent(vehicleStatus_);
29 this->requiresComponent(offboardKeepAlive_);
30
31 // Connect to disarmed signal - PX4 auto-disarms after landing
32 this->getStateMachine()->createSignalConnection(
33 vehicleStatus_->onDisarmed_, &CbLand::onLandedCallback, this);
34
35 // Disable offboard keepalive - land command uses its own mode
36 offboardKeepAlive_->disable();
37
38 RCLCPP_INFO(getLogger(), "CbLand: sending land command");
39 vehicleCommand_->land();
40}
