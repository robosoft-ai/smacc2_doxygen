27// A box standing on the table: its centre on the table plane and how it is turned.
28struct BoxFit
29{
30 double x = 0.0; // m, world frame
31 double y = 0.0;
32 double yawRad = 0.0; // of the box's x side about the vertical, in [0, pi/2) for a square box
33 double overlap = 0.0; // of the box's outline with the blob, 0 to 1 (intersection over union)
34 bool topOnly = false; // fitted to the top face alone: the sides were not part of the blob
35};
36
37struct BoxFitParams
38{
39 cv::Vec3d sizeM{0.035, 0.035, 0.035}; // along the box's x, y and the vertical
40 double searchM = 0.012; // the centre is looked for this far around the first guess
41 double stepM = 0.002;
42 double yawStepRad = 0.0436; // 2.5 degrees
43 double minOverlap = 0.6; // below this there is no fit
44 // A side face that looks like the table (or its shadow) is missing from the blob. When the
45 // whole outline fits worse than this, the top face alone is tried as well and the better kept.
46 double tryTopBelow = 0.85;
47 // For a box of one strong colour: blob pixels that are darker than the blob's middle
48 // lightness and have less than this fraction of its strong chroma (the 90th percentile) are
49 // left out of the fit. They are shadow, which near a coloured box takes on some of its colour.
50 // A pale but bright face of the box is kept. 0 uses the whole blob.
51 double minChromaFraction = 0.0;
52};
