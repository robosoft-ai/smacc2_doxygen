27// Preflight gate on the EKF heading. A vehicle sitting still on the ground must
28// report a constant heading; if the estimate is drifting, the yaw gyro bias
29// has not converged (or the magnetometer has been rejected) and a takeoff in
30// that state is unstable: with a large heading error the position loop pushes
31// the wrong way and the vehicle flips over (sm_cl_px4_mr_test_5 run 19: 186
32// degrees of drift in 30 s, crash 4 s after liftoff).
33//
34// Success when the heading moved less than maxDriftRadS * windowS over the
35// last windowS seconds (and, if expectedHeadingRad is finite, is also within
36// headingTolRad of it). Failure on timeout, with the drift rate in the log.
37struct HeadingStableParams
38{
39 double maxDriftRadS = 0.0087; // 0.5 deg/s
40 double windowS = 5.0;
41 double timeoutS = 60.0;
42 // known heading of the parked vehicle (NED, rad); NaN = do not check
43 float expectedHeadingRad = std::numeric_limits<float>::quiet_NaN();
44 float headingTolRad = 0.26f; // 15 deg
45};
46

49public:

51
52 void onEntry() override;
53 void onExit() override {}
54
55private:
56 HeadingStableParams params_;
57};
