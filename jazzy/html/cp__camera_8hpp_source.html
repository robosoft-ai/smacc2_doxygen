41// One camera: its latest image, and how its pixels relate to the world. The pose comes from
42// the transform tree at the time of the image, so a camera on a moving arm is handled the
43// same as a fixed one. Create one named component per camera.
44class CpCamera : public smacc2::ISmaccComponent
45{
46public:
47 explicit CpCamera(const CameraParams & params);
48 ~CpCamera() override;
49
50 void onInitialize() override;
51
52 const CameraParams & params() const { return params_; }
53
54 // the latest image; null until the first one arrives
55 FramePtr latestFrame() const;
56
57 // the camera as it was when the frame was taken; nothing while the calibration or the
58 // transform at that time is missing
59 std::optional<PinholeCamera> model(const Frame & frame) const;
60
61 // called on the executor thread with every image
62 smacc2::SmaccSignal<void(const FramePtr &)> onFrame_;
63
64private:
65 void onImage(const sensor_msgs::msg::Image::ConstSharedPtr msg);
66 void onInfo(const sensor_msgs::msg::CameraInfo::ConstSharedPtr msg);
67
68 CameraParams params_;
69 CpTransforms * transforms_ = nullptr;
