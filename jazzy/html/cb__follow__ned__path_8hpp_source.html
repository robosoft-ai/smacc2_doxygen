30// Streams a caller-supplied NED polyline through the carrot follower: one
31// continuous, leashed flight along a route of arbitrary vertices (a cave
32// passage, a survey line, a retrace of an earlier route) instead of a chain
33// of CbGoToLocation states or CbFollowWaypoints' setpoint jumps.
34//
35// The route is given at construction or via setPath() from the owning state's
36// runtimeConfigure (state machine thread, before onEntry). reachedCount() is
37// the number of route vertices the carrot has passed; a state reads it in its
38// onExit (state machine thread, updates barred) to record how far the vehicle
39// got, e.g. to retrace only the traversed prefix on an abort.
40class CbFollowNedPath : public CbPx4PathFollowerBase
41{
42public:
43 explicit CbFollowNedPath(std::vector<NedPoint> path = {}, PathFollowerParams follower = {});
44
45 void setPath(std::vector<NedPoint> path);
46 const std::vector<NedPoint> & path() const { return path_; }
47
48 // route vertices passed by the carrot, 0..path().size(); == path().size()
49 // once the path completed
50 size_t reachedCount() const;
51
52protected:
53 std::vector<NedPoint> buildPath(const NedPoint & /*current*/) override { return path_; }
54 const char * behaviorName() const override { return "CbFollowNedPath"; }
55
56private:
57 std::vector<NedPoint> path_;
58};
