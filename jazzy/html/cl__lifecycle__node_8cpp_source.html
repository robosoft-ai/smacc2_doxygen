46// Waits for the service, sends the request and waits for the response, checking `cancelled` and
47// rclcpp::ok() every 100 ms; both waits are unbounded but abandonable. Returns nullptr if abandoned.
48template <typename ServiceT>
49typename ServiceT::Response::SharedPtr callService(
50 const rclcpp::Logger & logger, typename rclcpp::Client<ServiceT>::SharedPtr client,
51 typename ServiceT::Request::SharedPtr request, const ClLifecycleNode::CancelPredicate & cancelled,
52 const std::string & what)
53{
54 const std::string serviceName = client->get_service_name();
55 auto abandoned = [&cancelled]() { return (cancelled && cancelled()) || !rclcpp::ok(); };
56
57 int laps = 0;
58 while (!client->wait_for_service(std::chrono::milliseconds(100)))
59 {
60 if (abandoned())
61 {
62 RCLCPP_INFO(
63 logger, "[ClLifecycleNode] %s abandoned while waiting for service %s", what.c_str(),
64 serviceName.c_str());
65 return nullptr;
66 }
67 if (++laps % 50 == 0)
68 {
69 RCLCPP_WARN(
70 logger, "[ClLifecycleNode] still waiting for service %s (%d s)", serviceName.c_str(),
71 laps / 10);
72 }
73 }
74
75 auto future = client->async_send_request(request);
76 laps = 0;
77 while (future.wait_for(std::chrono::milliseconds(100)) != std::future_status::ready)
78 {
79 if (abandoned())
80 {
81 client->remove_pending_request(future);
82 RCLCPP_INFO(
83 logger, "[ClLifecycleNode] %s abandoned while waiting for the response of %s", what.c_str(),
84 serviceName.c_str());
85 return nullptr;
86 }
87 if (++laps % 50 == 0)
88 {
89 RCLCPP_WARN(
90 logger, "[ClLifecycleNode] no response from %s after %d s", serviceName.c_str(), laps / 10);
91 }
92 }
93 return future.get();
94}

97bool ClLifecycleNode::changeState(uint8_t transition, const CancelPredicate & cancelled)
98{
99 auto logger = this->getNode()->get_logger();
100 auto request = std::make_shared<lifecycle_msgs::srv::ChangeState::Request>();
101 request->transition.id = transition;
102 auto response = callService<lifecycle_msgs::srv::ChangeState>(
103 logger, client_change_state_, request, cancelled, "transition " + std::to_string(transition));
104 if (!response)
105 {
106 return false;
107 }
108
109 const std::string serviceName = client_change_state_->get_service_name();
110 if (response->success)
111 {
112 RCLCPP_INFO(
113 logger, "[ClLifecycleNode] transition %d accepted by %s", transition, serviceName.c_str());
114 }
115 else
116 {
117 RCLCPP_ERROR(
118 logger, "[ClLifecycleNode] transition %d rejected by %s; no transition event will follow",
119 transition, serviceName.c_str());
120 }
121 return response->success;
122}

124bool ClLifecycleNode::getState(uint8_t & state, const CancelPredicate & cancelled)
125{
126 auto request = std::make_shared<lifecycle_msgs::srv::GetState::Request>();
127 auto response = callService<lifecycle_msgs::srv::GetState>(
128 this->getNode()->get_logger(), client_get_state_, request, cancelled, "get_state");
129 if (!response)
130 {
131 return false;
132 }
133 state = response->current_state.id;
134 return true;
135}

157bool ClLifecycleNode::shutdown(const CancelPredicate & cancelled)
158{
159 // Each primary state has its own shutdown transition id, so ask the node which state it is in.
160 using lifecycle_msgs::msg::State;
161 using lifecycle_msgs::msg::Transition;
162 uint8_t state;
163 if (!getState(state, cancelled))
164 {
165 return false;
166 }
167 switch (state)
168 {
169 case State::PRIMARY_STATE_UNCONFIGURED:
170 return changeState(Transition::TRANSITION_UNCONFIGURED_SHUTDOWN, cancelled);
171 case State::PRIMARY_STATE_INACTIVE:
172 return changeState(Transition::TRANSITION_INACTIVE_SHUTDOWN, cancelled);
173 case State::PRIMARY_STATE_ACTIVE:
174 return changeState(Transition::TRANSITION_ACTIVE_SHUTDOWN, cancelled);
175 default:
176 RCLCPP_ERROR(
177 this->getNode()->get_logger(),
178 "[ClLifecycleNode] %s is in lifecycle state %d, which has no shutdown transition",
179 nodeName_.c_str(), state);
180 return false;
181 }
182}
