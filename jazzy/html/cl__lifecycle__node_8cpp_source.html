44bool ClLifecycleNode::changeState(uint8_t transition, const CancelPredicate & cancelled)
45{
46 auto logger = this->getNode()->get_logger();
47 const std::string serviceName = client_change_state_->get_service_name();
48 auto abandoned = [&cancelled]() { return (cancelled && cancelled()) || !rclcpp::ok(); };
49
50 // The client is created in onInitialize() and the first request can follow within
51 // milliseconds, before the service has been discovered. A request sent before discovery has
52 // no reader and its future never completes, so wait for the server first. The wait is
53 // unbounded but abandonable: the predicate is checked every 100 ms.
54 int laps = 0;
55 while (!client_change_state_->wait_for_service(std::chrono::milliseconds(100)))
56 {
57 if (abandoned())
58 {
59 RCLCPP_INFO(
60 logger, "[ClLifecycleNode] transition %d abandoned while waiting for service %s",
61 transition, serviceName.c_str());
62 return false;
63 }
64 if (++laps % 50 == 0)
65 {
66 RCLCPP_WARN(
67 logger, "[ClLifecycleNode] still waiting for service %s (%d s)", serviceName.c_str(),
68 laps / 10);
69 }
70 }
71
72 auto request = std::make_shared<lifecycle_msgs::srv::ChangeState::Request>();
73 request->transition.id = transition;
74 auto future = client_change_state_->async_send_request(request);
75 laps = 0;
76 while (future.wait_for(std::chrono::milliseconds(100)) != std::future_status::ready)
77 {
78 if (abandoned())
79 {
80 client_change_state_->remove_pending_request(future);
81 RCLCPP_INFO(
82 logger, "[ClLifecycleNode] transition %d abandoned while waiting for the response of %s",
83 transition, serviceName.c_str());
84 return false;
85 }
86 if (++laps % 50 == 0)
87 {
88 RCLCPP_WARN(
89 logger, "[ClLifecycleNode] no response from %s after %d s", serviceName.c_str(), laps / 10);
90 }
91 }
92
93 const bool accepted = future.get()->success;
94 if (accepted)
95 {
96 RCLCPP_INFO(
97 logger, "[ClLifecycleNode] transition %d accepted by %s", transition, serviceName.c_str());
98 }
99 else
100 {
101 RCLCPP_ERROR(
102 logger, "[ClLifecycleNode] transition %d rejected by %s; no transition event will follow",
103 transition, serviceName.c_str());
104 }
105 return accepted;
106}
