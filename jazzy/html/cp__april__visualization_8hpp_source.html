64 void computeAggregatdMarker()
65 {
66 std::stringstream ss;
67
68 ss << "[CpAprilTagVisualization] computeAggregatdMarker" << std::endl;
69 cl_isaac_apriltag::ClIsaacApriltag* apriltagDetector = dynamic_cast<cl_isaac_apriltag::ClIsaacApriltag*>(this->owner_);
70
71 ss << "[CpAprilTagVisualization] getTagsWithinTime" << std::endl;
72 auto tags = apriltagDetector->getTagsWithinTime(0.5s);
73
74 //avergae apriltag position
75 double avgX = 0;
76 double avgY = 0;
77 double avgZ = 0;
78 for (auto &apriltag : tags)
79 {
80 avgX += apriltag.second.pose.position.x;
81 avgY += apriltag.second.pose.position.y;
82 avgZ += apriltag.second.pose.position.z;
83 }
84
85 if(tags.size() == 0)
86 {
87 RCLCPP_INFO_THROTTLE(getLogger(), *(getNode()->get_clock()), 1000, "[CpAprilTagVisualization] no apriltags detected to visualize");
88 return;
89 }
90
91 avgX /= tags.size();
92 avgY /= tags.size();
93 avgZ /= tags.size();
94
95 //substract some small forward offset from apriltag position for the goal position
96 avgX -= 0.15;
97
98 ss << "[CpAprilTagVisualization] detected " << tags.size() << " tags" << std::endl;
99 ss << "[CpAprilTagVisualization] avgX: " << avgX << ", avgY: " << avgY << ", avgZ: " << avgZ << std::endl;
100
101 tf2::Transform apriltagTransform;
102 apriltagTransform.setOrigin(tf2::Vector3(avgX, avgY, 0.0));
103 apriltagTransform.setRotation(tf2::Quaternion(0.0, 0.0, 0.0, 1.0));
104
105 globalApriltagTransform = apriltagTransform;
106
107 ss << "[CpAprilTagVisualization] apriltagTransform: " << apriltagTransform.getOrigin().x() << ", " << apriltagTransform.getOrigin().y() << ", " << apriltagTransform.getOrigin().z();
108 ss << "[CpAprilTagVisualization] globalApriltagTransform: " << globalApriltagTransform->getOrigin().x() << ", " << globalApriltagTransform->getOrigin().y() << ", " << globalApriltagTransform->getOrigin().z();
109 RCLCPP_INFO_THROTTLE(getLogger(), *(getNode()->get_clock()), 1000, ss.str().c_str());
110 }

112 virtual void update() override

123 visualization_msgs::msg::Marker marker;
124 marker.header.frame_id = "map";
125 marker.header.stamp = getNode()->now();
126 marker.ns = "apriltag";
127 marker.id = 0;
128 marker.action = visualization_msgs::msg::Marker::ADD;
129
130 //lifetim of 0.5s
131 marker.lifetime = rclcpp::Duration(0.5s);
132
133 // type sphere
134 marker.type = visualization_msgs::msg::Marker::SPHERE;
135
136 auto origin = globalApriltagTransform->getOrigin();
137 marker.pose.position.x = origin.x();
138 marker.pose.position.y = origin.y();
139 marker.pose.position.z = origin.z();
140
141 auto rotation = globalApriltagTransform->getRotation();
142 marker.pose.orientation.w = rotation.w();
143 marker.pose.orientation.y = rotation.y();
144 marker.pose.orientation.z = rotation.z();
145 marker.pose.orientation.x = rotation.x();
146
147 RCLCPP_INFO_THROTTLE(getLogger(), *(getNode()->get_clock()), 1000, "[CpAprilTagVisualization] marker position: %f, %f, %f", origin.x(), origin.y(), origin.z());
148
149 // red color
150 marker.color.r = 1.0;
151 marker.color.g = 0.0;
152 marker.color.b = 0.0;
153 marker.color.a = 1.0;
154
155 marker.scale.x = 0.1;
156 marker.scale.y = 0.1;
157 marker.scale.z = 0.1;
158
159 markerArray.markers.push_back(marker);
160
161 markerPublisher->publish(markerArray);
162 }
163 }
