193class CpKeyboardListener1 : public smacc2::ISmaccComponent
194{
195public:
196 CpKeyboardListener1() {}
197
198 virtual ~CpKeyboardListener1() {}
199
200 // Override the base class methods to call our setup
201 template <typename TOrthogonal, typename TClient>
202 void onComponentInitialization()
203 {
204 RCLCPP_INFO(getLogger(), "CpKeyboardListener1 initialization");
205 smacc2::client_core_components::CpTopicSubscriber<std_msgs::msg::UInt16> * subscriber;
206 this->requiresComponent(subscriber);
207
208 subscriber->onMessageReceived(&CpKeyboardListener1::onKeyboardMessage, this);
209
210 postEventKeyPress = [=](auto unicode_keychar)
211 {
212 // arrow codepoints (8592-8595) from the keyboard server's escape-sequence
213 // decoding: dispatch through the arrow signal/events, not the char path
214 switch (unicode_keychar.data)
215 {
216 case 8593:
217 this->postKeyEvent<EvKeyPressArrowUp<CpKeyboardListener1, TOrthogonal>>();
218 OnArrowPress_(ArrowKey::Up);
219 return;
220 case 8595:
221 this->postKeyEvent<EvKeyPressArrowDown<CpKeyboardListener1, TOrthogonal>>();
222 OnArrowPress_(ArrowKey::Down);
223 return;
224 case 8592:
225 this->postKeyEvent<EvKeyPressArrowLeft<CpKeyboardListener1, TOrthogonal>>();
226 OnArrowPress_(ArrowKey::Left);
227 return;
228 case 8594:
229 this->postKeyEvent<EvKeyPressArrowRight<CpKeyboardListener1, TOrthogonal>>();
230 OnArrowPress_(ArrowKey::Right);
231 return;
232 default:
233 break;
234 }
235
236 if (unicode_keychar.data > 255)
237 {
238 return; // other non-char codes have no dispatch
239 }
240
241 char character = (char)unicode_keychar.data;
242 RCLCPP_WARN(getLogger(), "detected keyboard: %c", character);
243
244 if (character == 'a')
245 this->postKeyEvent<EvKeyPressA<CpKeyboardListener1, TOrthogonal>>();
246 else if (character == 'b')
247 this->postKeyEvent<EvKeyPressB<CpKeyboardListener1, TOrthogonal>>();
248 else if (character == 'c')
249 this->postKeyEvent<EvKeyPressC<CpKeyboardListener1, TOrthogonal>>();
250 else if (character == 'd')
251 this->postKeyEvent<EvKeyPressD<CpKeyboardListener1, TOrthogonal>>();
252 else if (character == 'e')
253 this->postKeyEvent<EvKeyPressE<CpKeyboardListener1, TOrthogonal>>();
254 else if (character == 'f')
255 this->postKeyEvent<EvKeyPressF<CpKeyboardListener1, TOrthogonal>>();
256 else if (character == 'g')
257 this->postKeyEvent<EvKeyPressG<CpKeyboardListener1, TOrthogonal>>();
258 else if (character == 'h')
259 this->postKeyEvent<EvKeyPressH<CpKeyboardListener1, TOrthogonal>>();
260 else if (character == 'i')
261 this->postKeyEvent<EvKeyPressI<CpKeyboardListener1, TOrthogonal>>();
262 else if (character == 'j')
263 this->postKeyEvent<EvKeyPressJ<CpKeyboardListener1, TOrthogonal>>();
264 else if (character == 'k')
265 this->postKeyEvent<EvKeyPressK<CpKeyboardListener1, TOrthogonal>>();
266 else if (character == 'l')
267 this->postKeyEvent<EvKeyPressL<CpKeyboardListener1, TOrthogonal>>();
268 else if (character == 'm')
269 this->postKeyEvent<EvKeyPressM<CpKeyboardListener1, TOrthogonal>>();
270 else if (character == 'n')
271 this->postKeyEvent<EvKeyPressN<CpKeyboardListener1, TOrthogonal>>();
272 else if (character == 'o')
273 this->postKeyEvent<EvKeyPressO<CpKeyboardListener1, TOrthogonal>>();
274 else if (character == 'p')
275 this->postKeyEvent<EvKeyPressP<CpKeyboardListener1, TOrthogonal>>();
276 else if (character == 'q')
277 this->postKeyEvent<EvKeyPressQ<CpKeyboardListener1, TOrthogonal>>();
278 else if (character == 'r')
279 this->postKeyEvent<EvKeyPressR<CpKeyboardListener1, TOrthogonal>>();
280 else if (character == 's')
281 this->postKeyEvent<EvKeyPressS<CpKeyboardListener1, TOrthogonal>>();

283 this->postKeyEvent<EvKeyPressT<CpKeyboardListener1, TOrthogonal>>();

290 else if (character == 'x')
291 this->postKeyEvent<EvKeyPressX<CpKeyboardListener1, TOrthogonal>>();
292 else if (character == 'y')
293 this->postKeyEvent<EvKeyPressY<CpKeyboardListener1, TOrthogonal>>();
294 else if (character == 'z')
295 this->postKeyEvent<EvKeyPressZ<CpKeyboardListener1, TOrthogonal>>();
296
297 OnKeyPress_(character);
298 };
299 }

301 // This signal is created to be used in client behaviors
302 smacc2::SmaccSignal<void(char keypress)> OnKeyPress_;
303
304 // Arrow keypresses (decoded escape sequences) - char-typed OnKeyPress_ never

306 smacc2::SmaccSignal<void(ArrowKey arrow)> OnArrowPress_;

308 // This is the signal subscription method for client behaviors and other components
309 template <typename T>
310 void OnKeyPress(void (T::*callback)(char keypress), T * object)
311 {
312 this->getStateMachine()->createSignalConnection(OnKeyPress_, callback, object);
313 }
314
315 template <typename T>
316 void OnArrowPress(void (T::*callback)(ArrowKey arrow), T * object)
317 {
318 this->getStateMachine()->createSignalConnection(OnArrowPress_, callback, object);
319 }
320
321 void onKeyboardMessage(const std_msgs::msg::UInt16 & unicode_keychar)
322 {
323 postEventKeyPress(unicode_keychar);
324 }
325
326public:
327 template <typename TEv>
328 void postKeyEvent()
329 {
330 RCLCPP_WARN(
331 getLogger(), "CpKeyboardListener1 ev: %s",
332 smacc2::demangleSymbol(typeid(TEv).name()).c_str());
333 this->postEvent<TEv>();
334 }

336private:
