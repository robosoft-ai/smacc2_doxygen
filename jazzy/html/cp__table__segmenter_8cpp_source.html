26void CpTableSegmenter::onInitialize()
27{
28 const std::string path = this->params().referenceImagePath;
29 if (path.empty())
30 {
31 return;
32 }
33 const cv::Mat reference = cv::imread(path, cv::IMREAD_COLOR);
34 if (reference.empty())
35 {
36 RCLCPP_ERROR(
37 getLogger(),
38 "CpTableSegmenter: cannot read the reference image %s; segmenting by the table's own colour",
39 path.c_str());
40 return;
41 }
42 this->setReference(reference);
43 RCLCPP_INFO(
44 getLogger(), "CpTableSegmenter: reference image %s (%dx%d)", path.c_str(), reference.cols,
45 reference.rows);
46}

72cv::Mat CpTableSegmenter::foreground(const cv::Mat & bgr, const cv::Mat & tableMask) const
73{
74 cv::Mat reference;
75 {
76 std::lock_guard<std::mutex> lock(mutex_);
77 reference = reference_; // shares the pixels; they are never written after setReference
78 }
79 if (!reference.empty() && reference.size() == bgr.size())
80 {
81 return segmentAgainstReference(bgr, reference, tableMask, this->params());
82 }
83 return segmentForeground(bgr, tableMask, this->params());
84}
