170 // Typed subscribe helpers, one per signal. They go through the state machine's
171 // createSignalConnection, so a connection made by a state-scoped object (a client behavior
172 // subscribing with `this`) is tracked by that object's address and finalized when the object
173 // is disposed at its state's exit: a transition event arriving after the exit cannot reach a
174 // destroyed subscriber. Behaviors must use these from onStateOrthogonalAllocation (state
175 // machine thread), never a raw connect() on the signal members above (finding T) and never
176 // from an asynchronous onEntry/onExit thread (Async Thread Locking Rule).
177 template <typename T>
178 smacc2::SmaccSignalConnection onTransitionCreate(void (T::*callback)(), T * object)
179 {
180 return this->getStateMachine()->createSignalConnection(onTransitionCreate_, callback, object);
181 }
