42 flightPatternLawnmowerLaneCount(params_), params_.laneLength, params_.laneSpacing,
43 std::isnan(params_.laneHeading) ? current.yaw : params_.laneHeading,
44 params_.originIsCenter ? "centred" : "cornered",
