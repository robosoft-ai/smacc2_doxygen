37// Vision from colour images alone. The client only assembles the components: the cameras,
38// the table they look at, the segmenter and the place where a survey's result is kept.
39class ClRgb : public smacc2::ISmaccClient
40{
41public:
42 explicit ClRgb(const RgbParams & params);
43 ~ClRgb() override;
44
45 template <typename TOrthogonal, typename TClient>
46 void onComponentInitialization()
47 {
48 // the transform tree first: every camera resolves it in onInitialize
49 this->createComponent<CpTransforms, TOrthogonal, TClient>();
50 for (const auto & camera : params_.cameras)
51 {
52 this->createNamedComponent<CpCamera, TOrthogonal, TClient>(camera.name, camera);
53 }
54 this->createComponent<CpTablePlane, TOrthogonal, TClient>(params_.table);
55 this->createComponent<CpTableSegmenter, TOrthogonal, TClient>(params_.segmenter);
56 this->createComponent<CpSceneSurvey, TOrthogonal, TClient>();
57 }
58
59private:
60 RgbParams params_;
61};
