48struct ForwardObstacleGuardParams
49{
50 float halfAngleRad = 0.349f; // 20 deg cone about the sensor +x axis
51 float triggerRangeM = 3.0f; // too close below this ...
52 float clearRangeM = 4.0f; // ... cleared again above this (hysteresis)
53 float minRangeM = 0.6f; // ignore the airframe's own returns
54 // Ignore returns more than this far below the sensor (metres, sensor frame):
55 // the floor of a passage flown at low altitude must not count as an obstacle
56 // ahead; anything reaching up to within this of the vehicle still does. 0 disables.
57 float floorCutoffM = 0.8f;
58 int minHits = 10; // cone points below triggerRangeM per cloud to count
59 int triggerClouds = 2; // consecutive triggering clouds before "too close"
60 int clearClouds = 10; // consecutive clear clouds before "cleared"
61 // Clouds captured while the vehicle yaws faster than this (rad/s, derived from
62 // successive /fmu/out/vehicle_attitude headings) are ignored: the cone is
63 // sweeping the walls, not looking along the direction of travel (e.g. the
64 // 180 deg turn at the start of a retrace). 0 disables the gate.
65 float maxYawRateRadS = 0.35f;
66 // Evaluate the cone in a level frame: each cloud is rotated by the vehicle's
67 // roll and pitch (from /fmu/out/vehicle_attitude) so the cone axis is the
68 // true heading and floorCutoffM is true height below the sensor. Without it
69 // a nose-down pitch tilts the floor up into the cone.
70 bool levelFrame = true;
71 // The cloud is in a camera OPTICAL frame (z forward, x right, y down - e.g. a
72 // depth_image_proc point cloud) rather than the sensor frame assumed above
73 // (x forward, y left, z up): remap every point before the cone test.
74 bool opticalFrameInput = false;
75};

77// Watches a PointCloud2 (sensor frame) for returns inside a forward cone and
78// raises / clears an obstacle flag with hysteresis. "Forward" is the sensor's
79// +x axis: with the lidar mounted level and every leg flown in
80// YawMode::TANGENT, body x is the direction of travel while cruising. The
81// cone is not the velocity direction during vertical legs, holds, or the
82// 180-degree turn at the start of a retrace - keep the trigger range below
83// half the narrowest passage width so those sweeps do not trip it.
84//
85// Requires a CpTopicSubscriber<sensor_msgs::msg::PointCloud2> on the same
86// client (ClGenericSensor<PointCloud2> creates one). Subscribes to PX4's
87// /fmu/out/vehicle_attitude itself (level frame + yaw-rate gate). The sensor must be mounted level with the body (x
88// forward, z up). onCloud runs on the ROS executor; it only emits SmaccSignals.
89class CpForwardObstacleGuard : public smacc2::ISmaccComponent
90{
91public:

94
