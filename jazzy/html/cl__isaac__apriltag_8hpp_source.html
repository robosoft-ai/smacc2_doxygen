85 std::lock_guard<std::mutex> lock(detectedAprilTagsMapPoseMutex_);
86 std::map<std::string, geometry_msgs::msg::PoseStamped> ret;

88 auto now = getNode()->now();
89
90
91 std::stringstream ss;
92 for (auto &apriltag : detectedAprilTagsMapPose_)
93 {
94 auto tagstamp = rclcpp::Time(apriltag.second.header.stamp);
95 rclcpp::Duration ellapsed = now - tagstamp; ;
96
97 ss << "[ClIsaacApriltag] AprilTag: " << apriltag.first << " Ellapsed: " << ellapsed.seconds() << " Stamp: " << tagstamp.seconds() << " now " << now.seconds() << std::endl;
98 if ( ellapsed < duration)
99 {
100 ret.insert(apriltag);
101 ss << "[SELECTD]" << std::endl;
102 }
103 else

105 ss << "[NOT SELECTED]" << std::endl;
106 }
107 ss << std::endl;
108
109 }
110
111 RCLCPP_INFO_THROTTLE(getLogger(), *(getNode()->get_clock()), 1000, "%s", ss.str().c_str());
112 return ret;
113 }

115private:
116 void onAprilTagMessageCallback(const isaac_ros_apriltag_interfaces::msg::
117 AprilTagDetectionArray::SharedPtr msg) {
118
119 std::lock_guard<std::mutex> lock(detectedAprilTagsMapPoseMutex_);
120
121 std::stringstream ss;
122 for (auto &detection : msg->detections) {
123 std::string apriltag_frameid =
124 detection.family + ":" + std::to_string(detection.id);
125 ss << "[ClIsaacApriltag] AprilTag detected: " << apriltag_frameid
126 << std::endl;
127
128 // get map position using tfListener
129 geometry_msgs::msg::TransformStamped transformStampedGlobal;
130
131 try {
132 transformStampedGlobal = tfBuffer_->lookupTransform(
133 "map", apriltag_frameid, msg->header.stamp);
134 } catch (tf2::TransformException &ex) {
135 RCLCPP_ERROR(getLogger(), "%s", ex.what());
136 continue;
137 }
138 // transform to pose
139 geometry_msgs::msg::PoseStamped poseStamped;
140 poseStamped.header = transformStampedGlobal.header;
141 poseStamped.pose.position.x = transformStampedGlobal.transform.translation.x;
142 poseStamped.pose.position.y = transformStampedGlobal.transform.translation.y;
143 poseStamped.pose.position.z = transformStampedGlobal.transform.translation.z;
144 poseStamped.pose.orientation = transformStampedGlobal.transform.rotation;
145
146 detectedAprilTagsMapPose_[apriltag_frameid] = poseStamped;
147
148 ss << "[ClIsaacApriltag] new AprilTag detected: " << apriltag_frameid << " at "
149 << poseStamped.pose.position.x << ", " << poseStamped.pose.position.y << ", " << poseStamped.pose.position.z;
150 }
151 RCLCPP_INFO_STREAM_THROTTLE(getLogger(), *(getNode()->get_clock()), 1000,
152 ss.str());
153 onAprilTagDetection_(msg);
154 }
