37void CbAbortNavigation::onEntry()
38{
39 // connected before the request, so the result of the cancelled goal cannot be missed
40 goalFinished_ = false;
41 nav2ActionInterface_->onNavigationCancelled(&CbAbortNavigation::onNavigationResult, this);
42 nav2ActionInterface_->onNavigationAborted(&CbAbortNavigation::onNavigationResult, this);
43 nav2ActionInterface_->onNavigationSucceeded(&CbAbortNavigation::onNavigationResult, this);
44
45 if (!nav2ActionInterface_->cancelNavigation())
46 {
47 RCLCPP_INFO(getLogger(), "[CbAbortNavigation] no navigation goal in flight");
48 this->postSuccessEvent();
49 return;
50 }
51
52 const auto deadline = std::chrono::steady_clock::now() + std::chrono::seconds(5);
53 while (!goalFinished_)
54 {
55 // the state was left first: no event
56 if (this->isShutdownRequested()) return;
57 if (std::chrono::steady_clock::now() >= deadline)
58 {
59 RCLCPP_ERROR(getLogger(), "[CbAbortNavigation] no result for the cancelled goal after 5 s");
60 this->postFailureEvent();
61 return;
62 }
63 std::this_thread::sleep_for(std::chrono::milliseconds(10));
64 }
65
66 RCLCPP_INFO(getLogger(), "[CbAbortNavigation] the cancelled goal has finished");
67 this->postSuccessEvent();
68}
