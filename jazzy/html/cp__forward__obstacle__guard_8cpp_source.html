39void CpForwardObstacleGuard::onInitialize()
40{
41 this->requiresComponent(subscriber_);
42 if (subscriber_ == nullptr)
43 {
44 RCLCPP_ERROR(
45 getLogger(),
46 "CpForwardObstacleGuard: no CpTopicSubscriber<PointCloud2> on this client - guard inactive");
47 return;
48 }
49 subscriber_->onMessageReceived(&CpForwardObstacleGuard::onCloud, this);
50 if (params_.levelFrame || params_.maxYawRateRadS > 0.0f)
51 {
52 attitudeSub_ = this->getNode()->create_subscription<px4_msgs::msg::VehicleAttitude>(
53 "/fmu/out/vehicle_attitude", rclcpp::SensorDataQoS(),
54 std::bind(&CpForwardObstacleGuard::onAttitude, this, std::placeholders::_1));
55 }
56 RCLCPP_INFO(
57 getLogger(),
58 "CpForwardObstacleGuard: subscribed - cone %.0f deg, trigger < %.1f m, clear > %.1f m, "
59 "%d hits, %d/%d clouds, yaw-rate gate %.2f rad/s, floor cutoff %.1f m, level frame %s",
60 params_.halfAngleRad * 180.0 / M_PI, params_.triggerRangeM, params_.clearRangeM,
61 params_.minHits, params_.triggerClouds, params_.clearClouds, params_.maxYawRateRadS,
62 params_.floorCutoffM, params_.levelFrame ? "on" : "off");
63}

78void CpForwardObstacleGuard::onAttitude(const px4_msgs::msg::VehicleAttitude::SharedPtr msg)
79{
80 // yaw rate by finite difference of the NED heading (PX4 does not bridge
81 // vehicle_angular_velocity by default); light EMA against quantisation
82 const double w = msg->q[0], x = msg->q[1], y = msg->q[2], z = msg->q[3];
83 const double yaw = std::atan2(2.0 * (w * z + x * y), 1.0 - 2.0 * (y * y + z * z));
84 if (lastYawStampUs_ != 0 && msg->timestamp > lastYawStampUs_)
85 {
86 const double dt = static_cast<double>(msg->timestamp - lastYawStampUs_) * 1e-6;
87 double d = yaw - lastYaw_;
88 d = std::atan2(std::sin(d), std::cos(d));
89 if (dt > 1e-4 && dt < 0.5)
90 {
91 const float rate = static_cast<float>(d / dt);
92 yawRate_ = 0.5f * yawRate_.load() + 0.5f * rate;
93 }
94 }
95 lastYaw_ = yaw;
96 lastYawStampUs_ = msg->timestamp;
97
98 std::lock_guard<std::mutex> lock(attitudeMutex_);
99 attitudeQ_ = {msg->q[0], msg->q[1], msg->q[2], msg->q[3]};
100 haveAttitude_ = true;
101}

103bool CpForwardObstacleGuard::levelRotation(std::array<float, 9> & r) const
104{
105 std::array<float, 4> q;
106 {
107 std::lock_guard<std::mutex> lock(attitudeMutex_);
108 if (!haveAttitude_)
109 {
110 return false;
111 }
112 q = attitudeQ_;
113 }
114 // body FLU -> ENU (px4_ros_com constants), then strip the yaw so the result
115 // is the roll/pitch-only rotation into a level, heading-aligned frame
116 const tf2::Quaternion kNedEnuQ(0.70710678118654752, 0.70710678118654752, 0.0, 0.0);
117 const tf2::Quaternion kFrdFluQ(1.0, 0.0, 0.0, 0.0);
118 const tf2::Quaternion qNedFrd(q[1], q[2], q[3], q[0]);
119 tf2::Quaternion qEnuFlu = kNedEnuQ * qNedFrd * kFrdFluQ;
120 qEnuFlu.normalize();
121 const tf2::Matrix3x3 m(qEnuFlu);
122 const double yaw = std::atan2(m[1][0], m[0][0]);
123 tf2::Matrix3x3 unyaw;
124 unyaw.setRPY(0.0, 0.0, -yaw);
125 const tf2::Matrix3x3 level = unyaw * m;
126 for (int i = 0; i < 3; ++i)
127 {
128 for (int j = 0; j < 3; ++j)
129 {
130 r[i * 3 + j] = static_cast<float>(level[i][j]);
131 }
132 }
133 return true;
134}

136void CpForwardObstacleGuard::onCloud(const sensor_msgs::msg::PointCloud2 & msg)
137{
138 if (params_.maxYawRateRadS > 0.0f && std::fabs(yawRate_.load()) > params_.maxYawRateRadS)
139 {
140 // turning: the cone is sweeping sideways; keep the current flag, count nothing
141 if (++gatedClouds_ == 1)
142 {
143 RCLCPP_INFO(
144 getLogger(), "CpForwardObstacleGuard: yawing at %.2f rad/s - cone gated", yawRate_.load());
145 }
146 return;
147 }
148 if (gatedClouds_ > 0)
149 {
150 RCLCPP_INFO(
151 getLogger(), "CpForwardObstacleGuard: cone active again after %d gated clouds", gatedClouds_);
152 gatedClouds_ = 0;
153 triggerStreak_ = 0;
154 }
155
156 int hits = 0;
157 float minRange = std::numeric_limits<float>::infinity();
158
159 std::array<float, 9> r{};
160 const bool level = params_.levelFrame && levelRotation(r);
161
162 sensor_msgs::PointCloud2ConstIterator<float> ix(msg, "x");
163 sensor_msgs::PointCloud2ConstIterator<float> iy(msg, "y");
164 sensor_msgs::PointCloud2ConstIterator<float> iz(msg, "z");
165 for (; ix != ix.end(); ++ix, ++iy, ++iz)
166 {
167 float x = *ix;
168 float y = *iy;
169 float z = *iz;
170 if (!std::isfinite(x) || !std::isfinite(y) || !std::isfinite(z))
171 {
172 continue; // gz reports no-return as inf (is_dense = false)
173 }
174 if (params_.opticalFrameInput)
175 {
176 // optical (z forward, x right, y down) -> sensor (x forward, y left, z up)
177 const float ox = x, oy = y, oz = z;
178 x = oz;
179 y = -ox;
180 z = -oy;
181 }
182 if (level)
183 {
184 const float lx = r[0] * x + r[1] * y + r[2] * z;
185 const float ly = r[3] * x + r[4] * y + r[5] * z;
186 const float lz = r[6] * x + r[7] * y + r[8] * z;
187 x = lx;
188 y = ly;
189 z = lz;
190 }
191 if (params_.floorCutoffM > 0.0f && z < -params_.floorCutoffM)
192 {
193 continue; // the floor, not something in the way
194 }
195 const float r = std::sqrt(x * x + y * y + z * z);
196 if (r < params_.minRangeM || r > params_.clearRangeM)
197 {
198 continue;
199 }
200 if (x / r < cosHalfAngle_)
201 {
202 continue; // outside the forward cone
203 }
204 if (r < minRange)
205 {
206 minRange = r;
207 }
208 if (r < params_.triggerRangeM)
209 {
210 ++hits;
211 }
212 }
213
214 lastMinRange_ = minRange;
215 lastHits_ = hits;
216
217 const bool triggering = hits >= params_.minHits;
218 // cleared only when nothing in the cone is below the clear range at all
219 const bool clear = !(minRange < params_.clearRangeM);
220
221 if (!tooClose_)
222 {
223 triggerStreak_ = triggering ? triggerStreak_ + 1 : 0;
224 if (triggerStreak_ >= params_.triggerClouds)
225 {
226 tooClose_ = true;
227 clearStreak_ = 0;
228 RCLCPP_WARN(
229 getLogger(), "CpForwardObstacleGuard: OBSTACLE min range %.2f m (%d hits in cone)",
230 minRange, hits);
231 onObstacleTooClose_();
232 }
233 }
234 else
235 {
236 clearStreak_ = clear ? clearStreak_ + 1 : 0;
237 if (clearStreak_ >= params_.clearClouds)
238 {
239 tooClose_ = false;
240 triggerStreak_ = 0;
241 RCLCPP_INFO(getLogger(), "CpForwardObstacleGuard: cleared (min range %.2f m)", minRange);
242 onObstacleCleared_();
243 }
244 }
245}
