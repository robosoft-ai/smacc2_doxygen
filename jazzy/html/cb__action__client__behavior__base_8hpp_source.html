36// Generic base for client behaviors that send one goal to a ROS 2 action server
37// (through a CpActionClient<TAction> component) and finish when the action
38// finishes. Derived behaviors fill a Goal in onEntry() and call sendGoal();
39// the base turns the action result into the state-scoped behavior events
40// EvCbSuccess<TDerived, TOrthogonal> / EvCbFailure<TDerived, TOrthogonal>.
41//
42// Lifecycle contract (see the async-thread locking rule in CLAUDE.md):
43// - components are resolved and result signals wired in
44// onStateOrthogonalAllocation, on the state machine thread. Never from the
45// asynchronous onEntry/onExit threads.
46// - a derived class that declares its own onStateOrthogonalAllocation MUST
47// chain to this one (and this one chains to SmaccAsyncClientBehavior's, which
48// installs the event-posting functions - skipping the chain leaves them empty
49// and postSuccessEvent() throws std::bad_function_call).
50// - no machine-scoped events are posted by this base: transition tables react
51// to the EvCb* behavior events, which cannot leak across state transitions.
52template <typename TAction>
53class CbActionClientBehaviorBase : public smacc2::SmaccAsyncClientBehavior
54{
55public:
56 using Goal = typename TAction::Goal;
57 using GoalHandle = rclcpp_action::ClientGoalHandle<TAction>;
58 using WrappedResult = typename GoalHandle::WrappedResult;
59 using Feedback = typename TAction::Feedback;
60 using ActionClientComponent = smacc2::client_core_components::CpActionClient<TAction>;
61
62 template <typename TOrthogonal, typename TSourceObject>
63 void onStateOrthogonalAllocation()
64 {
65 if (actionClientName_.empty())
66 {
67 this->requiresComponent(actionClient_, ComponentRequirement::HARD);
68 }
69 else
70 {
71 // target a specific named instance when the client holds several
72 // CpActionClient components of the same action type
73 this->requiresComponent(actionClientName_, actionClient_, ComponentRequirement::HARD);
74 }
75
76 if (!resultConnectionsInitialized_ && actionClient_ != nullptr)
77 {
78 actionClient_->onSucceeded(&CbActionClientBehaviorBase::dispatchActionSuccess, this);
79 actionClient_->onAborted(&CbActionClientBehaviorBase::dispatchActionAbort, this);
80 actionClient_->onCancelled(&CbActionClientBehaviorBase::dispatchActionAbort, this);
81 actionClient_->onFeedback(&CbActionClientBehaviorBase::dispatchActionFeedback, this);
82 resultConnectionsInitialized_ = true;
83 }
84
85 smacc2::SmaccAsyncClientBehavior::onStateOrthogonalAllocation<TOrthogonal, TSourceObject>();
86 }
87
88 virtual ~CbActionClientBehaviorBase() {}
89
90 // If the state exits while a goal is still in flight (e.g. a keyboard or
91 // timeout transition), cancel it: unlike bt_navigator navigation - where the
92 // next goal implicitly preempts - each behavior/docking server action keeps
93 // executing an abandoned goal, leaving the robot moving under a command
94 // nobody owns. Derived classes overriding onExit must chain to this.
95 void onExit() override
96 {
97 if (goalInFlight_)
98 {
99 RCLCPP_WARN(
100 getLogger(), "[%s] State exited with the action goal still in flight - cancelling",
101 getName().c_str());
102 cancelGoal();
103 }
104 }
105
106protected:
107 // Sends the goal and waits (in the calling asynchronous onEntry thread) for the
108 // server's goal response. Returns true if the goal was accepted. On a
109 // not-ready server, a rejected goal, or a response timeout it posts the
110 // failure event and returns false - the caller can simply return from
111 // onEntry(). This closes the silent-hang gap of a rejected/unanswered goal:
112 // CpActionClient installs no goal_response handling, and a rejected goal
113 // never invokes the result callback.
114 bool sendGoal(Goal & goal)
115 {
116 if (actionClient_ == nullptr || !actionClient_->isServerReady())
117 {
118 RCLCPP_ERROR(
119 getLogger(), "[%s] Action server not available, cannot send goal", getName().c_str());
120 this->postFailureEvent();
121 return false;
122 }
123
124 goalActivitySeen_ = false;
125 auto goalHandleFuture = actionClient_->sendGoal(goal);
126 {
127 std::lock_guard<std::mutex> lock(ownGoalMutex_);
128 ownGoalHandleFuture_ = goalHandleFuture;
129 }
130
131 // wait for goal acceptance in short slices so a state exit is honored
132 auto deadline = std::chrono::steady_clock::now() + goalResponseTimeout_;
133 while (!this->isShutdownRequested() && std::chrono::steady_clock::now() < deadline)
134 {
135 // Feedback or a result arriving proves the goal was accepted even if
136 // the goal-response future hasn't been serviced yet: under spin_some, a
137 // high-rate feedback stream (NavigateToPose feeds back continuously
138 // from acceptance) can starve the action client's response processing
139 // for the whole execution. Rejected goals produce no feedback, so their
140 // response resolves the future promptly.
141 if (goalActivitySeen_)
142 {
143 goalInFlight_ = true;
144 return true;
145 }
146
147 if (goalHandleFuture.wait_for(std::chrono::milliseconds(50)) == std::future_status::ready)
148 {
149 if (goalHandleFuture.get() != nullptr)
150 {
151 goalInFlight_ = true; // accepted; the result signals will finish the behavior
152 return true;
153 }
154
155 RCLCPP_ERROR(getLogger(), "[%s] Goal was rejected by the action server", getName().c_str());
156 this->postFailureEvent();
157 return false;
158 }
159 }
160
161 if (!this->isShutdownRequested())
162 {
163 // No response, no feedback, no result within the window: ambiguous, and
164 // observed only under executor starvation with an accepted goal - a
165 // genuine rejection resolves the future in milliseconds. Assume the
166 // goal is running and let the result signals finish the behavior
167 // (fire-and-forget, the pre-template behavior), rather than failing a
168 // healthy mission.
169 RCLCPP_WARN(
170 getLogger(),
171 "[%s] No goal response from the action server after %ld ms; assuming the goal was "
172 "accepted (a rejection responds promptly) - result signals will finish this behavior",
173 getName().c_str(), static_cast<long>(goalResponseTimeout_.count()));
174 goalInFlight_ = true;
175 }
176 return true;
177 }
178
179 void cancelGoal()
180 {
181 if (actionClient_ != nullptr)
182 {
183 actionClient_->cancelGoal();
184 }
185 }
186
187 // Result handlers connected by onStateOrthogonalAllocation. Base
188 // implementations record the result code and post the behavior events;
189 // derived classes may override to customize result handling.
190 virtual void onActionSuccess(const WrappedResult & result)
191 {
192 actionResult_ = result.code;
193 RCLCPP_INFO(getLogger(), "[%s] Action succeeded, propagating success event", getName().c_str());
194 this->postSuccessEvent();
195 }
196
197 virtual void onActionAbort(const WrappedResult & result)
198 {
199 actionResult_ = result.code;
200 RCLCPP_INFO(getLogger(), "[%s] Action failed, propagating failure event", getName().c_str());
201 this->postFailureEvent();
202 }
203
204 // optional: override to consume action feedback (distance traveled etc.)
205 virtual void onActionFeedback(const Feedback & /*feedback*/) {}
206
207 // set from the derived constructor to bind to a named CpActionClient
208 // instance; empty binds the first of matching type. Must be set before
209 // onStateOrthogonalAllocation runs (i.e. NOT in runtimeConfigure, which
210 // executes after allocation)
211 void setActionClientName(std::string name) { actionClientName_ = std::move(name); }
212
213 ActionClientComponent * actionClient_ = nullptr;
214 std::string actionClientName_;
215
216 rclcpp_action::ResultCode actionResult_ = rclcpp_action::ResultCode::UNKNOWN;
217
218 // how long sendGoal waits for the server to accept/reject the goal
219 std::chrono::milliseconds goalResponseTimeout_ = std::chrono::milliseconds(10000);
220
221private:
222 // trampolines wired to the component signals: clear the in-flight flag
223 // regardless of what a derived handler override does, then dispatch to the
224 // overridable virtuals
225 void dispatchActionSuccess(const WrappedResult & result)
226 {
227 if (!isOwnGoal(result)) return;
228 goalActivitySeen_ = true;
229 goalInFlight_ = false;
230 this->onActionSuccess(result);
231 }
232
233 void dispatchActionAbort(const WrappedResult & result)
234 {
235 if (!isOwnGoal(result)) return;
236 goalActivitySeen_ = true;
237 goalInFlight_ = false;
238 this->onActionAbort(result);
239 }
240
241 void dispatchActionFeedback(const Feedback & feedback)
242 {
243 goalActivitySeen_ = true;
244 this->onActionFeedback(feedback);
245 }
246
247 // The component's result signals fire for every goal sent through it, so a result of the previous
248 // state's goal (typically its cancellation) can arrive while this behavior's own goal runs.
249 bool isOwnGoal(const WrappedResult & result)
250 {
251 std::shared_future<typename GoalHandle::SharedPtr> future;
252 {
253 std::lock_guard<std::mutex> lock(ownGoalMutex_);
254 future = ownGoalHandleFuture_;
255 }
256 // rclcpp_action resolves the goal response before it can deliver that goal's result.
257 const bool own = future.valid() &&
258 future.wait_for(std::chrono::seconds(0)) == std::future_status::ready &&
259 future.get() != nullptr && future.get()->get_goal_id() == result.goal_id;
260 if (!own)
261 {
262 RCLCPP_INFO(
263 getLogger(), "[%s] Ignoring the result of a goal this behavior did not send",
264 getName().c_str());
265 }
266 return own;
267 }
268
269 bool resultConnectionsInitialized_ = false;
270 std::mutex ownGoalMutex_;
271 std::shared_future<typename GoalHandle::SharedPtr> ownGoalHandleFuture_;
272 std::atomic<bool> goalInFlight_{false};
273 std::atomic<bool> goalActivitySeen_{false};
274};
