30void CbObstacleGuard::onEntry()
31{
32 this->requiresComponent(guard_);
33 if (guard_ == nullptr)
34 {
35 RCLCPP_ERROR(
36 getLogger(), "CbObstacleGuard: CpForwardObstacleGuard not found - obstacle events disabled");
37 return;
38 }
39 // start from a clean slate: whatever the cone saw before this behavior
40 // existed (the ground while on the pad, walls swept during a turn) is not a
41 // detection; a real obstacle re-triggers within a couple of clouds
42 guard_->reset();
43 guard_->onObstacleTooClose(&CbObstacleGuard::onTooClose, this);
44 guard_->onObstacleCleared(&CbObstacleGuard::onCleared, this);
45 RCLCPP_INFO(getLogger(), "CbObstacleGuard: armed");
46}
