39bool replace(std::string & str, const std::string & from, const std::string & to)
40{
41 size_t start_pos = str.find(from);
42 if (start_pos == std::string::npos) return false;
43 str.replace(start_pos, from.length(), to);
44 return true;
45}

47std::string replace_back(
48 std::string roottype, std::vector<std::pair<std::string, std::string>> & orderedtypesdict)
49{
50 while (roottype.find("$") != std::string::npos)
51 {
52 for (auto & t : orderedtypesdict)
53 {
54 auto & tkey = t.first;
55 auto & tval = t.second;
56
57 replace(roottype, tkey, tval);
58 }
59 }
60
61 return roottype;
62}

71TypeInfo::Ptr TypeInfo::getTypeInfoFromString(std::string inputtext)
72{
73 auto it = typeInfoDatabase.find(inputtext);
74
75 if (it != typeInfoDatabase.end()) return typeInfoDatabase[inputtext];
76
77 bool ok = false;
78 int typecount = 0;
79 std::string originalinputtext = inputtext;
80 std::map<std::string, std::string> typesdict;
81 std::map<std::string, std::string> typesdict_content;
82
83 // ITERATE on the current string to replace internal template types with ids (ie $Tx) replacements
84 // until plain template type
85
86 while (!ok)
87 {
88 const char * simpletypeRE = "[^\\<\\>,\\s]+\\<[^\\<\\>]+\\>";
89
90 // locate moste outer template
91 std::smatch matches;
92 std::regex regex(simpletypeRE); // matches words beginning by "sub"
93
94 std::regex_search(inputtext, matches, regex);
95
96 // checking ending condition
97 if (matches.size() == 0)
98 {
99 if (typesdict.size() == 0)
100 {
101 auto tkey = "$T" + std::to_string(typecount);
102 typesdict[tkey] = inputtext;
103 }
104 break;
105 }
106 else
107 {
108 // find and replace all coincidences typestr to tokenid
109 int i = 0;
110 for (auto & m : matches)
111 {
112 std::string tstr = m;
113 auto tkey = "$T" + std::to_string(typecount);
114 replace(inputtext, tstr, tkey);
115
116 typecount++;
117 typesdict[tkey] = tstr;
118
119 i++;
120 }
121 }
122 }
123
124 // ---------- SORT by token key to replace back in a single pass --------
125 std::vector<std::pair<std::string, std::string>> orderedTypedict;
126 for (const auto & item : typesdict)
127 {
128 orderedTypedict.emplace_back(item);
129 }
130
131 std::sort(
132 orderedTypedict.begin(), orderedTypedict.end(),
133 [](auto & a, auto & b) { return std::stoi(a.first.substr(2)) > std::stoi(b.first.substr(2)); });
134 // -----------------------------------------------------------------
135
136 std::set<std::string> allbasetypes;
137 for (auto & typeentry : orderedTypedict /*typesdict*/)
138 {
139 auto flat = typeentry.second;
140
141 size_t startindex = flat.find("<");
142 size_t endindex = flat.find(">");
143 if (startindex != std::string::npos)
144 {
145 flat = flat.substr(
146 startindex + 1,
147 endindex - startindex - 1); // gets the list of template parameters in csv format
148
149 typesdict_content[typeentry.first] = flat;
150 std::vector<std::string> localbasetypes;
151
152 std::istringstream iss(flat);
153 std::string token;
154 while (std::getline(iss, token, ','))
155 {
156 boost::trim(token);
157 localbasetypes.push_back(token);
158 }
159
160 for (auto & b : localbasetypes)
161 {
162 size_t found = b.find("$");
163
164 if (found == std::string::npos)
165 {
166 allbasetypes.insert(b);
167 }
168 }
169 }
170
171 //refresh
172 for (auto & b : allbasetypes)
173 {
174 typesdict[b] = b;
175 }
176 }
177 // append leaf types
178 for (auto & b : allbasetypes)
179 {
180 orderedTypedict.push_back({b, b});
181 }
182
183 std::vector<TypeInfo::Ptr> types;
184 std::vector<std::string> tokens;
185
186 for (auto t : orderedTypedict) // must be ordered. to avoid issue, ie: $11 to be replaced in $T14
187 {
188 auto & tkey = t.first;
189 auto & tval = t.second;
190 auto finaltype = replace_back(tval, orderedTypedict);
191 auto tinfo = std::make_shared<TypeInfo>(tkey, tval, finaltype);
192 types.push_back(tinfo);
193 tokens.push_back(tkey);
194 }
195
196 // --------- FINDING THE CURRENT TYPE -------------
197 TypeInfo::Ptr roottype = nullptr;
198 for (auto & t : types)
199 {
200 if (t->finaltype == originalinputtext)
201 {
202 roottype = t;
203 break;
204 }
205 }
206
207 if (roottype == nullptr && globalNh_ != nullptr)
208 {
209 std::stringstream ss;
210 ss << "---------------" << std::endl;
211 ss << "TYPE STRING WALKER, type was not properly interpreted: " << std::endl
212 << " - " << inputtext.c_str() << std::endl
213 << " - " << originalinputtext << std::endl;
214
215 auto strmsg = ss.str();
216 RCLCPP_WARN_STREAM(globalNh_->get_logger(), strmsg);
217 }
218
219 // Replacing back tokens in strings to get the whole final name
220 for (size_t i = 0; i < types.size(); i++)
221 {
222 auto t = types[i];
223 auto ttoken = tokens[i];
224
225 std::vector<std::pair<int, TypeInfo::Ptr>> unorderedTemplateParameters;
226
227 auto codedtypecopy = typesdict_content[ttoken];
228
229 for (auto & t2 : types)
230 {
231 if (t == t2) continue;
232
233 auto index = codedtypecopy.find(t2->tkey);
234 if (index != std::string::npos)
235 {
236 auto pair = std::make_pair(index, t2); // this line is important for the order of templates
237 unorderedTemplateParameters.push_back(pair);
238 replace(codedtypecopy, t2->tkey, ""); // consume token
239 //std::cout << "codedtypecopy: " << codedtypecopy << std::endl;
240 }
241 }
242
243 std::sort(
244 unorderedTemplateParameters.begin(), unorderedTemplateParameters.end(),
245 [](auto & a, auto & b) -> bool { return a.first <= b.first; });
246
247 //RCLCPP_DEBUG_STREAM(nh_->get_logger(),"------------------");
248 //RCLCPP_DEBUG_STREAM(nh_->get_logger(),"CREATING TYPE:" << t->getFullName());
249
250 for (auto & item : unorderedTemplateParameters)
251 {
252 //RCLCPP_DEBUG_STREAM(nh_->get_logger()," - template parameter: " << item.second->getFullName());
253 t->templateParameters.push_back(item.second);
254 }
255 //RCLCPP_DEBUG_STREAM(nh_->get_logger(),"------------------");
256 }
257
258 //RCLCPP_DEBUG_STREAM(nh_->get_logger(),"ADDING TYPE TO DATABASE: " << inputtext);
259 //RCLCPP_DEBUG_STREAM(nh_->get_logger(),"Current Database");
260 for (auto & en : typeInfoDatabase)
261 {
262 if (globalNh_ != nullptr) RCLCPP_DEBUG_STREAM(globalNh_->get_logger(), "- " << en.first);
263 }
264
265 typeInfoDatabase[originalinputtext] = roottype;
266
267 if (roottype == nullptr && globalNh_ != nullptr)
268 {
269 std::stringstream ss;
270 ss << "---------------" << std::endl;
271 ss << "TYPE STRING WALKER, type was not properly interpreted: " << std::endl
272 << " - " << inputtext.c_str() << std::endl
273 << " - " << originalinputtext << std::endl;
274
275 ss << "---- current type (ordered types)" << std::endl;
276
277 for (auto & en : orderedTypedict)
278 {
279 ss << "- " << en.first << ": " << en.second << std::endl;
280 }
281 ss << "---------------" << std::endl;
282
283 ss << "------current type types --------" << std::endl;
284
285 for (auto & t : types)
286 {
287 ss << t->finaltype << std::endl;
288 }
289
290 ss << "----total typeinfo database" << std::endl;
291
292 for (auto & en : typeInfoDatabase)
293 {
294 ss << "- " << en.first << ": " << en.second << std::endl;
295 }
296
297 auto strmsg = ss.str();
298 RCLCPP_WARN_STREAM(globalNh_->get_logger(), strmsg);
299 }
300
301 return roottype;
302}
