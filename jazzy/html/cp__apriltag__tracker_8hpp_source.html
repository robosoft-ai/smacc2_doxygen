77 std::map<std::string, geometry_msgs::msg::PoseStamped> getTagsWithinTime(
78 rclcpp::Duration duration)
79 {
80 std::lock_guard<std::mutex> lock(mutex_);
81 std::map<std::string, geometry_msgs::msg::PoseStamped> ret;
82
83 auto now = getNode()->now();
84
85 std::stringstream ss;
86 for (auto & apriltag : detectedAprilTagsMapPose_)
87 {
88 auto tagstamp = rclcpp::Time(apriltag.second.header.stamp);
89 rclcpp::Duration elapsed = now - tagstamp;
90
91 ss << "[CpAprilTagTracker] AprilTag: " << apriltag.first << " Elapsed: " << elapsed.seconds()
92 << " Stamp: " << tagstamp.seconds() << " now " << now.seconds() << std::endl;
93 if (elapsed < duration)
94 {
95 ret.insert(apriltag);
96 ss << "[SELECTED]" << std::endl;
97 }
98 else
99 {
100 ss << "[NOT SELECTED]" << std::endl;
101 }
102 ss << std::endl;
103 }
104
105 RCLCPP_INFO_THROTTLE(getLogger(), *(getNode()->get_clock()), 1000, "%s", ss.str().c_str());
106 return ret;
107 }

155 void onAprilTagMessage(const AprilTagDetectionArray & msg)
156 {
157 std::lock_guard<std::mutex> lock(mutex_);
158
159 std::stringstream ss;
160 for (auto & detection : msg.detections)
161 {
162 std::string apriltag_frameid = detection.family + ":" + std::to_string(detection.id);
163 ss << "[CpAprilTagTracker] AprilTag detected: " << apriltag_frameid << std::endl;
164
165 // Get map position using TF lookup
166 geometry_msgs::msg::TransformStamped transformStampedGlobal;
167
168 try
169 {
170 transformStampedGlobal =
171 tfBuffer_->lookupTransform(targetFrame_, apriltag_frameid, msg.header.stamp);
172 }
173 catch (tf2::TransformException & ex)
174 {
175 RCLCPP_WARN_THROTTLE(
176 getLogger(), *(getNode()->get_clock()), 1000,
177 "[CpAprilTagTracker] TF lookup failed for %s: %s", apriltag_frameid.c_str(), ex.what());
178 continue;
179 }
180
181 // Transform to pose
182 geometry_msgs::msg::PoseStamped poseStamped;
183 poseStamped.header = transformStampedGlobal.header;
184 poseStamped.pose.position.x = transformStampedGlobal.transform.translation.x;
185 poseStamped.pose.position.y = transformStampedGlobal.transform.translation.y;
186 poseStamped.pose.position.z = transformStampedGlobal.transform.translation.z;
187 poseStamped.pose.orientation = transformStampedGlobal.transform.rotation;
188
189 detectedAprilTagsMapPose_[apriltag_frameid] = poseStamped;
190
191 ss << "[CpAprilTagTracker] new AprilTag detected: " << apriltag_frameid << " at "
192 << poseStamped.pose.position.x << ", " << poseStamped.pose.position.y << ", "
193 << poseStamped.pose.position.z;
194 }
195 RCLCPP_INFO_STREAM_THROTTLE(getLogger(), *(getNode()->get_clock()), 1000, ss.str());
196
197 // Emit signal with SharedPtr for downstream subscribers
198 auto msgPtr = std::make_shared<AprilTagDetectionArray>(msg);
199 onAprilTagDetection_(msgPtr);
200 }
