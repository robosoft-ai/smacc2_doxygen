29class CbConfigure : public smacc2::SmaccAsyncClientBehavior

31public:
32 CbConfigure() {}

34
35 template <typename TOrthogonal, typename TSourceObject>
36 void onStateOrthogonalAllocation()
37 {
38 smacc2::SmaccAsyncClientBehavior::onStateOrthogonalAllocation<TOrthogonal, TSourceObject>();
39
40 this->requiresClient(lifecycleNodeClient_);
41 this->requiresComponent(eventMonitor_);
42 // State-scoped signal connections, made on the state machine thread. Each is tracked by this
43 // behavior's address and finalized when the behavior is disposed at state exit, so a
44 // transition event that arrives after the exit cannot call into a destroyed behavior. The
45 // raw connect() calls this replaces were never disconnected (finding T).
46 eventMonitor_->onTransitionOnConfigureSuccess(&CbConfigure::onTransitionSuccess, this);
47 eventMonitor_->onTransitionOnConfigureFailure(&CbConfigure::onTransitionFailure, this);
48 eventMonitor_->onTransitionOnConfigureError(&CbConfigure::onTransitionFailure, this);
49 }
50
51 virtual void onEntry() override
52 {
53 // Runs on this behavior's thread. The wait inside the client ends when the request is
54 // answered, when this behavior is asked to finish (state exit, shutdown) or when ROS shuts
55 // down. If the request was abandoned nothing is posted: the state is already leaving. If the
56 // node rejected it, no transition event will ever arrive for this attempt, so the behavior
57 // reports the failure itself.
58 const bool accepted =
59 lifecycleNodeClient_->configure([this]() { return this->isShutdownRequested(); });
60 if (!accepted && !this->isShutdownRequested())
61 {
62 this->postFailureEvent();
63 }
64 }
65
66 void onTransitionSuccess()
67 {
68 RCLCPP_INFO(getLogger(), "[CbConfigure] Configure transition succeeded");
69 this->postSuccessEvent();
70 }
71
72 void onTransitionFailure()
73 {
74 RCLCPP_WARN(getLogger(), "[CbConfigure] Configure transition failed or errored");
75 this->postFailureEvent();
76 }
77
78private:
79 ClLifecycleNode * lifecycleNodeClient_;
80 CpLifecycleEventMonitor * eventMonitor_;
81};
