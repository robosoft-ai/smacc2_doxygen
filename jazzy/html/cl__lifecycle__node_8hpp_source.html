203 void onInitialize() override;

205 // Predicate polled between the 100 ms waits inside changeState(); returning true abandons the
206 // request. Behaviors pass their isShutdownRequested(), so a state exit or a process shutdown
207 // ends the wait instead of leaving the behavior thread blocked.
208 using CancelPredicate = std::function<bool()>;
209
210 // @brief trigger a lifecycle transition on the node. Waits until the change_state service has
211 // been discovered, sends the request and waits for the response, checking `cancelled` and
212 // rclcpp::ok() between waits. Returns true if the server accepted the transition. Returns false
213 // if the wait was abandoned (predicate true or ROS shut down) or the server rejected the
214 // transition; a rejection is logged at ERROR, and no transition event will follow it.
215 bool changeState(uint8_t transition, const CancelPredicate & cancelled = CancelPredicate());
216
217 // @brief execute this method to trigger the configure transition
218 bool configure(const CancelPredicate & cancelled = CancelPredicate());
219
220 // @brief execute this method to trigger the activate transition
221 bool activate(const CancelPredicate & cancelled = CancelPredicate());
222
223 // @brief execute this method to trigger the deactivate transition
224 bool deactivate(const CancelPredicate & cancelled = CancelPredicate());
225
226 // @brief execute this method to trigger the cleanup transition
227 bool cleanup(const CancelPredicate & cancelled = CancelPredicate());
228
229 // @brief execute this method to trigger the shutdown transition
230 bool shutdown(const CancelPredicate & cancelled = CancelPredicate());
231
232 // @brief execute this method to trigger the destroy transition
233 void destroy();

238 template <typename TOrthogonal, typename TSourceObject>
239 void onStateOrthogonalAllocation()
240 {
241 // Create and register component with orthogonal
242 if (!eventMonitor_)
243 {
244 eventMonitor_ =
245 this->template createComponent<CpLifecycleEventMonitor, TOrthogonal, ClLifecycleNode>(
246 nodeName_);
247 }
248
249 // Configure component for this orthogonal
250 eventMonitor_->template onStateOrthogonalAllocation<TOrthogonal, ClLifecycleNode>();
251 }
252
253private:
254 // Phase 2: Service clients (will be migrated to CpServiceClient components in future enhancement)
255 rclcpp::Client<lifecycle_msgs::srv::GetState>::SharedPtr client_get_state_;
256 rclcpp::Client<lifecycle_msgs::srv::ChangeState>::SharedPtr client_change_state_;
257
258 std::string nodeName_;
259
260 // Phase 3: Event monitor component
261 CpLifecycleEventMonitor * eventMonitor_;

263 const std::string node_change_state_topic = "/change_state";
264
265 lifecycle_msgs::msg::TransitionEvent::SharedPtr lastTransitionEvent_;
