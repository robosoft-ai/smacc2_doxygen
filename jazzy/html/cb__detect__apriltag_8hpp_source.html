72 void onAprilTagDetected(
73 const isaac_ros_apriltag_interfaces::msg::AprilTagDetectionArray::SharedPtr & msg)
74 {
75 for (auto & detection : msg->detections)
76 {
77 std::string apriltag_frame_id = detection.family + ":" + std::to_string(detection.id);
78
79 // Check mission state via component (NOT direct client fields!)
80 if (!missionState_->isTagVisited(apriltag_frame_id))
81 {
82 RCLCPP_INFO_STREAM(
83 getLogger(), "[CbDetectAprilTag] new unvisited AprilTag detected: " << detection.id);
84
85 // Select tag via component method (NOT direct client field assignment!)
86 missionState_->selectTag(apriltag_frame_id);
87 postEvAprilTagDetected_();
88 }
89 else
90 {
91 RCLCPP_INFO_STREAM_THROTTLE(
92 getLogger(), *(getNode()->get_clock()), 2000,
93 "[CbDetectAprilTag] Skipping AprilTag already detected: " << apriltag_frame_id);
94 }
95 }
96 }
97};
