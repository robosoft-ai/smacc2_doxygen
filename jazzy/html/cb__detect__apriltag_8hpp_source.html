52 void onAprilTagDetected(const isaac_ros_apriltag_interfaces::msg::
53 AprilTagDetectionArray::SharedPtr &msg) {
54 for (auto &detection : msg->detections) {
55
56 std::string apriltag_frame_id =
57 detection.family + ":" + std::to_string(detection.id);
58 if (std::find(detectorClient->visitedWorkingAreas_.begin(),
59 detectorClient->visitedWorkingAreas_.end(),
60 apriltag_frame_id) ==
61 detectorClient->visitedWorkingAreas_.end()) {
62 RCLCPP_INFO_STREAM(
63 getLogger(), "[CbDetectAprilTag] new unvisited AprilTag detected: "
64 << detection.id);
65 detectorClient->selectedVisitTagId_ = apriltag_frame_id;
66 postEvAprilTagDetected_();
67 } else {
68 RCLCPP_INFO_STREAM_THROTTLE(
69 getLogger(), *(getNode()->get_clock()), 2000,
70 "[CbDetectAprilTag] Skipping AprilTag already detected: "
71 << apriltag_frame_id);
72 }
73 }
74 }
75};
