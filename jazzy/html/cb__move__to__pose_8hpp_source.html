41// Moves the tip link of its client's planning group to a pose: one plan, then its execution.
42// Without parameters it does nothing and posts no event, so one can stand idle on an orthogonal.
43class CbMoveToPose : public CbMoveit2zClientBehaviorBase
44{
45public:
46 CbMoveToPose() = default;
47
48 explicit CbMoveToPose(const MoveToPoseParams & params) : params_(params) {}
49
50 void setParams(const MoveToPoseParams & params) { params_ = params; }
51
52 void onEntry() override
53 {
54 if (!params_)
55 {
56 RCLCPP_INFO(getLogger(), "[CbMoveToPose] no pose given: idle");
57 return;
58 }
59 if (cpMotionPlanner_ == nullptr)
60 {
61 RCLCPP_ERROR(getLogger(), "[CbMoveToPose] the client has no CpMotionPlanner");
62 this->postMotionFailure();
63 return;
64 }
65
66 PlanningOptions options;
67 options.planningTime = params_->planningTime;
68 options.maxVelocityScaling = params_->velocityScaling;
69 options.maxAccelerationScaling = params_->accelerationScaling;
70 const std::optional<std::string> tipLink =
71 params_->tipLink.empty() ? std::nullopt : std::optional<std::string>(params_->tipLink);
72
73 const auto planned = cpMotionPlanner_->planToPose(params_->pose, tipLink, options);
74 if (!planned.success)
75 {
76 RCLCPP_WARN(getLogger(), "[CbMoveToPose] no plan: %s", planned.errorMessage.c_str());
77 this->postMotionFailure();
78 return;
79 }
80
81 bool executed = false;
82 if (cpTrajectoryExecutor_ != nullptr)
83 {
84 ExecutionOptions executionOptions;
85 executionOptions.trajectoryName = this->getName();
86 executed = cpTrajectoryExecutor_->executePlan(planned.plan, executionOptions).success;
87 }
88 else
89 {
90 executed = cpMoveGroup_->moveGroupClientInterface->execute(planned.plan) ==
91 moveit::core::MoveItErrorCode::SUCCESS;
92 }
93
94 if (executed)
95 {
96 this->postMotionSuccess();
97 }
98 else
99 {
100 RCLCPP_WARN(getLogger(), "[CbMoveToPose] the execution failed");
101 this->postMotionFailure();
102 }
103 }
104
105private:
106 std::optional<MoveToPoseParams> params_;
107};
