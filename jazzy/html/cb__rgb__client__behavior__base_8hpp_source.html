32// Base for cl_rgb client behaviors. The components are resolved and the completion signals
33// wired in onStateOrthogonalAllocation, on the state machine thread; onEntry, which runs on
34// the behavior's own thread, only computes (see the async thread locking rule in CLAUDE.md).
35//
36// Derived behaviors:
37// - override wireCompletionSignals() to connect a component signal, when they finish on one
38// - finish through postRgbSuccess()/postRgbFailure(), which post once however often called
39// - may arm a watchdog with setTimeout(); it is checked from update() and posts failure
40class CbRgbClientBehaviorBase : public smacc2::SmaccAsyncClientBehavior,
41 public smacc2::ISmaccUpdatable
42{
43public:
44 explicit CbRgbClientBehaviorBase(const std::string & cameraName) : cameraName_(cameraName) {}
45
46 ~CbRgbClientBehaviorBase() override {}
47
48 template <typename TOrthogonal, typename TSourceObject>
49 void onStateOrthogonalAllocation()
50 {
51 this->requiresComponent(cameraName_, camera_, smacc2::ComponentRequirement::SOFT);
52 this->requiresComponent(table_, smacc2::ComponentRequirement::SOFT);
53 this->requiresComponent(segmenter_, smacc2::ComponentRequirement::SOFT);
54 this->requiresComponent(survey_, smacc2::ComponentRequirement::SOFT);
55
56 this->wireCompletionSignals();
57
58 smacc2::SmaccAsyncClientBehavior::onStateOrthogonalAllocation<TOrthogonal, TSourceObject>();
59 }
60
61 // wall time after which the behavior posts failure; never when not called
62 void setTimeout(std::chrono::milliseconds timeout) { timeoutMs_ = timeout.count(); }
63
64 void update() override
65 {
66 const int64_t timeoutMs = timeoutMs_.load();
67 if (timeoutMs <= 0 || completed_)
68 {
69 return;
70 }
71 const auto now = std::chrono::steady_clock::now();
72 if (!watchdogStart_)
73 {
74 watchdogStart_ = now;
75 return;
76 }
77 if (now - *watchdogStart_ > std::chrono::milliseconds(timeoutMs))
78 {
79 RCLCPP_ERROR(
80 getLogger(), "[%s] Timed out after %ld ms without completing - posting failure",
81 getName().c_str(), static_cast<long>(timeoutMs));
82 this->postRgbFailure();
83 }
84 }
85
86protected:
87 virtual void wireCompletionSignals() {}
88
89 void postRgbSuccess()
90 {
91 if (!completed_.exchange(true))
92 {
93 this->postSuccessEvent();
94 }
95 }
96
97 void postRgbFailure()
98 {
99 if (!completed_.exchange(true))
100 {
101 this->postFailureEvent();
102 }
103 }
104
105 bool completed() const { return completed_; }
106
107 std::string cameraName_;
108 CpCamera * camera_ = nullptr;
109 CpTablePlane * table_ = nullptr;
110 CpTableSegmenter * segmenter_ = nullptr;
111 CpSceneSurvey * survey_ = nullptr;
112
113private:
114 std::atomic<bool> completed_{false};
115 std::atomic<int64_t> timeoutMs_{0}; // 0 = watchdog disabled
116 std::optional<std::chrono::steady_clock::time_point> watchdogStart_;
117};
