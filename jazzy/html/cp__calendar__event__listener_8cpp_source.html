75void CpCalendarEventListener::update()
76{
77 if (!initialized_ || !poller_)
78 {
79 return;
80 }
81
82 std::lock_guard<std::mutex> lock(mutex_);
83
84 // Check each watch pattern against current events
85 for (const auto & watch : watches_)
86 {
87 for (const auto & event : current_events_)
88 {
89 if (!matchesPattern(event, watch))
90 {
91 continue;
92 }
93
94 // Check for start trigger
95 if (watch.trigger_on_start && shouldTriggerStart(event, watch))
96 {
97 std::string key = getTriggeredKey(event, "start");
98
99 // Check if already triggered (unless continuous mode)
100 if (watch.continuous || triggered_events_.find(key) == triggered_events_.end())
101 {
102 triggered_events_.insert(key);
103
104 RCLCPP_INFO(
105 getLogger(), "[CpCalendarEventListener] Event start triggered: %s",
106 event.title.c_str());
107
108 // Emit signals
109 onEventStarted_(event);
110 if (postEventStartedEvent_)
111 {
112 postEventStartedEvent_(event);
113 }
114 }
115 }
116
117 // Check for end trigger
118 if (watch.trigger_on_end && shouldTriggerEnd(event, watch))
119 {
120 std::string key = getTriggeredKey(event, "end");
121
122 if (watch.continuous || triggered_events_.find(key) == triggered_events_.end())
123 {
124 triggered_events_.insert(key);
125
126 RCLCPP_INFO(
127 getLogger(), "[CpCalendarEventListener] Event end triggered: %s", event.title.c_str());
128
129 onEventEnded_(event);
130 if (postEventEndedEvent_)
131 {
132 postEventEndedEvent_(event);
133 }
134 }
135 }
136 }
137 }
138}

140void CpCalendarEventListener::onAgendaUpdated(const std::vector<CalendarEvent> & events)
141{
142 std::lock_guard<std::mutex> lock(mutex_);
143 current_events_ = events;
144
145 // Check for newly detected events matching patterns
146 for (const auto & watch : watches_)
147 {
148 for (const auto & event : events)
149 {
150 if (matchesPattern(event, watch))
151 {
152 std::string key = getTriggeredKey(event, "detected");
153
154 if (watch.continuous || triggered_events_.find(key) == triggered_events_.end())
155 {
156 triggered_events_.insert(key);
157
158 RCLCPP_DEBUG(
159 getLogger(), "[CpCalendarEventListener] Event detected matching pattern '%s': %s",
160 watch.pattern.c_str(), event.title.c_str());
161
162 onEventDetected_(event, watch.pattern);
163 if (postEventDetectedEvent_)
164 {
165 postEventDetectedEvent_(event, watch.pattern);
166 }
167 }
168 }
169 }
170 }

173bool CpCalendarEventListener::matchesPattern(
174 const CalendarEvent & event, const EventWatch & watch) const
175{
176 if (watch.use_regex)
177 {
178 try
179 {
180 boost::regex regex(watch.pattern, boost::regex::icase);
181 return boost::regex_search(event.title, regex);
182 }
183 catch (const boost::regex_error & e)
184 {
185 RCLCPP_ERROR(
186 getLogger(), "[CpCalendarEventListener] Invalid regex pattern '%s': %s",
187 watch.pattern.c_str(), e.what());
188 return false;
189 }
190 }
191 else
192 {
193 // Case-insensitive exact substring match
194 std::string title_lower = event.title;
195 std::string pattern_lower = watch.pattern;
196 std::transform(title_lower.begin(), title_lower.end(), title_lower.begin(), ::tolower);
197 std::transform(pattern_lower.begin(), pattern_lower.end(), pattern_lower.begin(), ::tolower);
198 return title_lower.find(pattern_lower) != std::string::npos;
199 }
200}

202bool CpCalendarEventListener::shouldTriggerStart(
203 const CalendarEvent & event, const EventWatch & watch) const
204{
205 auto now = std::chrono::system_clock::now();
206
207 // Calculate the trigger time (start_time - minutes_before)
208 auto trigger_time = event.start_time - std::chrono::minutes(watch.minutes_before);
209
210 // We trigger if:
211 // 1. Current time is at or past the trigger time
212 // 2. Event hasn't ended yet
213 // 3. We're within a reasonable window (not more than minutes_before + 1 minute past start)
214 auto max_trigger_window = event.start_time + std::chrono::minutes(1);
215
216 return now >= trigger_time && now < event.end_time && now <= max_trigger_window;
217}
218
219bool CpCalendarEventListener::shouldTriggerEnd(
220 const CalendarEvent & event, const EventWatch & /* watch */) const
221{
222 auto now = std::chrono::system_clock::now();
223
224 // Trigger end if:
225 // 1. Current time is at or past the end time
226 // 2. We're within a reasonable window (1 minute past end)
227 auto max_trigger_window = event.end_time + std::chrono::minutes(1);
228
229 return now >= event.end_time && now <= max_trigger_window;
230}
