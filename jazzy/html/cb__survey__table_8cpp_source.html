28void CbSurveyTable::onEntry()
29{
30 if (camera_ == nullptr || table_ == nullptr || segmenter_ == nullptr || survey_ == nullptr)
31 {
32 RCLCPP_ERROR(
33 getLogger(),
34 "CbSurveyTable: a component is missing (camera '%s' %d, table %d, segmenter %d, survey %d)",
35 params_.camera.c_str(), camera_ != nullptr, table_ != nullptr, segmenter_ != nullptr,
36 survey_ != nullptr);
37 this->postRgbFailure();
38 return;
39 }
40
41 // The latest image, with the camera as it stood when the image was taken.
42 FramePtr frame;
43 std::optional<PinholeCamera> model;
44 const auto deadline =
45 std::chrono::steady_clock::now() + std::chrono::duration<double>(params_.waitS);
46 while (!this->isShutdownRequested())
47 {
48 frame = camera_->latestFrame();
49 if (frame)
50 {
51 model = camera_->model(*frame);
52 if (model)
53 {
54 break;
55 }
56 }
57 if (std::chrono::steady_clock::now() > deadline)
58 {
59 RCLCPP_ERROR(
60 getLogger(),
61 "CbSurveyTable: no usable image of '%s' after %.1f s (image %d, calibration and pose %d)",
62 params_.camera.c_str(), params_.waitS, frame != nullptr, model.has_value());
63 this->postRgbFailure();
64 return;
65 }
66 std::this_thread::sleep_for(std::chrono::milliseconds(20));
67 }
68 if (!frame || !model)
69 {
70 this->postRgbFailure();
71 return;
72 }
73
74 const auto start = std::chrono::steady_clock::now();
75 const cv::Mat region = table_->regionMask(*model, frame->image.size());
76 const cv::Mat foreground = segmenter_->foreground(frame->image, region);
77 auto survey = std::make_shared<Survey>();
78 survey->stamp = frame->stamp;
79 survey->camera = params_.camera;
80 survey->model = *model;
81 survey->sequence = frame->sequence;
82 survey->blobs = findBlobs(frame->image, foreground, *model, table_->params(), params_.blobs);
83 const double ms =
84 std::chrono::duration<double, std::milli>(std::chrono::steady_clock::now() - start).count();
85
86 RCLCPP_INFO(
87 getLogger(), "CbSurveyTable: %zu blob(s) in image %lu of '%s' (%.1f ms)", survey->blobs.size(),
88 static_cast<unsigned long>(frame->sequence), params_.camera.c_str(), ms);
89 for (const auto & blob : survey->blobs)
90 {
91 RCLCPP_INFO(
92 getLogger(),
93 "CbSurveyTable: blob %d at x %+.3f y %+.3f m, %.0f x %.0f mm, axis %+.0f deg, %d px, "
94 "colour L %.0f a %+.0f b %+.0f",
95 blob.id, blob.tablePoint[0], blob.tablePoint[1], blob.lengthM * 1000.0, blob.widthM * 1000.0,
96 blob.axisRad * 180.0 / CV_PI, blob.areaPx, blob.meanLab[0], blob.meanLab[1], blob.meanLab[2]);
97 }
98
99 const bool enough = static_cast<int>(survey->blobs.size()) >= params_.minBlobs;
100 survey_->store(survey);
101 if (enough)
102 {
103 this->postRgbSuccess();
104 }
105 else
106 {
107 RCLCPP_WARN(
108 getLogger(), "CbSurveyTable: %zu blob(s), %d needed - posting failure", survey->blobs.size(),
109 params_.minBlobs);
110 this->postRgbFailure();
111 }
112}
