54 bool isTagVisited(const std::string & tag_frame_id) const

65 void markTagVisited(const std::string & tag_frame_id)

67 std::lock_guard<std::mutex> lock(mutex_);
68 if (
69 std::find(visitedWorkingAreas_.begin(), visitedWorkingAreas_.end(), tag_frame_id) ==
70 visitedWorkingAreas_.end())
71 {
72 visitedWorkingAreas_.push_back(tag_frame_id);
73 RCLCPP_INFO_STREAM(
74 getLogger(), "[CpAprilTagMissionState] Marked tag as visited: " << tag_frame_id);
75 }
76 }

86 }
