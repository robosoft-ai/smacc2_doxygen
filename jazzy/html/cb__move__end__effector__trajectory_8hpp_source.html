110 virtual void onEntry() override
111 {
112 this->requiresClient(movegroupClient_);
113
114 // Get optional components for visualization
115 CpTrajectoryVisualizer * trajectoryVisualizer = nullptr;
116 this->requiresComponent(trajectoryVisualizer, false); // Optional component
117
118 RCLCPP_INFO_STREAM(getLogger(), "[" << getName() << "] Generating end effector trajectory");
119
120 this->generateTrajectory();
121
122 if (this->endEffectorTrajectory_.size() == 0)
123 {
124 RCLCPP_WARN_STREAM(
125 getLogger(), "[" << smacc2::demangleSymbol(typeid(*this).name())
126 << "] No points in the trajectory. Skipping behavior.");
127 return;
128 }
129
130 // Use CpTrajectoryVisualizer if available, otherwise use legacy marker system
131 if (trajectoryVisualizer != nullptr)
132 {
133 RCLCPP_INFO_STREAM(
134 getLogger(),
135 "[" << getName() << "] Setting trajectory visualization using CpTrajectoryVisualizer.");
136 trajectoryVisualizer->setTrajectory(this->endEffectorTrajectory_, "trajectory");
137 }
138 else
139 {
140 RCLCPP_INFO_STREAM(
141 getLogger(), "[" << getName()
142 << "] Creating markers (legacy mode - consider adding "
143 "CpTrajectoryVisualizer component).");
144 this->createMarkers();
145 }
146
147 moveit_msgs::msg::RobotTrajectory computedTrajectory;
148
149 RCLCPP_INFO_STREAM(getLogger(), "[" << getName() << "] Computing joint space trajectory.");
150
151 auto errorcode = computeJointSpaceTrajectory(computedTrajectory);
152
153 bool trajectoryGenerationSuccess = errorcode == ComputeJointTrajectoryErrorCode::SUCCESS;
154
155 CpTrajectoryHistory * trajectoryHistory;
156 this->requiresComponent(trajectoryHistory);
157
158 if (!trajectoryGenerationSuccess)
159 {
160 RCLCPP_INFO_STREAM(
161 getLogger(), "[" << this->getName() << "] Incorrect trajectory. Posting failure event.");
162 if (trajectoryHistory != nullptr)
163 {
164 moveit_msgs::msg::MoveItErrorCodes error;
165 error.val = moveit_msgs::msg::MoveItErrorCodes::NO_IK_SOLUTION;
166 trajectoryHistory->pushTrajectory(this->getName(), computedTrajectory, error);
167 }
168
169 movegroupClient_->postEventMotionExecutionFailed();
170 this->postFailureEvent();
171
172 if (errorcode == ComputeJointTrajectoryErrorCode::JOINT_TRAJECTORY_DISCONTINUITY)
173 {
174 this->postJointDiscontinuityEvent(computedTrajectory);
175 }
176 else if (errorcode == ComputeJointTrajectoryErrorCode::INCORRECT_INITIAL_STATE)
177 {
178 this->postIncorrectInitialStateEvent(computedTrajectory);
179 }
180 return;
181 }
182 else
183 {
184 if (trajectoryHistory != nullptr)
185 {
186 moveit_msgs::msg::MoveItErrorCodes error;
187 error.val = moveit_msgs::msg::MoveItErrorCodes::SUCCESS;
188 trajectoryHistory->pushTrajectory(this->getName(), computedTrajectory, error);
189 }
190
191 this->executeJointSpaceTrajectory(computedTrajectory);
192 }
193
194 // handle finishing events
195 }

202protected:
203 ComputeJointTrajectoryErrorCode computeJointSpaceTrajectory(
204 moveit_msgs::msg::RobotTrajectory & computedJointTrajectory)
205 {
206 // Try to use CpJointSpaceTrajectoryPlanner component (preferred)
207 CpJointSpaceTrajectoryPlanner * trajectoryPlanner = nullptr;
208 this->requiresComponent(trajectoryPlanner, false); // Optional component
209
210 if (trajectoryPlanner != nullptr)
211 {
212 // Use component-based trajectory planner (preferred)
213 RCLCPP_INFO(
214 getLogger(),
215 "[CbMoveEndEffectorTrajectory] Using CpJointSpaceTrajectoryPlanner component for IK "
216 "trajectory generation");
217
218 JointTrajectoryOptions options;
219 if (tipLink_ && !tipLink_->empty())
220 {
221 options.tipLink = *tipLink_;
222 }
223 if (allowInitialTrajectoryStateJointDiscontinuity_)
224 {
225 options.allowInitialDiscontinuity = *allowInitialTrajectoryStateJointDiscontinuity_;
226 }
227
228 auto result = trajectoryPlanner->planFromWaypoints(endEffectorTrajectory_, options);
229
230 if (result.success)
231 {
232 computedJointTrajectory = result.trajectory;
233 RCLCPP_INFO(
234 getLogger(),
235 "[CbMoveEndEffectorTrajectory] IK trajectory generation succeeded (via "
236 "CpJointSpaceTrajectoryPlanner)");
237 return ComputeJointTrajectoryErrorCode::SUCCESS;
238 }
239 else
240 {
241 computedJointTrajectory = result.trajectory; // Still return partial trajectory
242 RCLCPP_WARN(
243 getLogger(),
244 "[CbMoveEndEffectorTrajectory] IK trajectory generation failed (via "
245 "CpJointSpaceTrajectoryPlanner): %s",
246 result.errorMessage.c_str());
247
248 // Map error codes
249 switch (result.errorCode)
250 {
251 case JointTrajectoryErrorCode::INCORRECT_INITIAL_STATE:
252 return ComputeJointTrajectoryErrorCode::INCORRECT_INITIAL_STATE;
253 case JointTrajectoryErrorCode::JOINT_TRAJECTORY_DISCONTINUITY:
254 return ComputeJointTrajectoryErrorCode::JOINT_TRAJECTORY_DISCONTINUITY;
255 default:
256 return ComputeJointTrajectoryErrorCode::JOINT_TRAJECTORY_DISCONTINUITY;
257 }
258 }
259 }
260 else
261 {
262 // Fallback to legacy IK trajectory generation
263 RCLCPP_WARN(
264 getLogger(),
265 "[CbMoveEndEffectorTrajectory] CpJointSpaceTrajectoryPlanner component not available, "
266 "using legacy IK trajectory generation (consider adding CpJointSpaceTrajectoryPlanner "
267 "component)");
268
269 // LEGACY IMPLEMENTATION (keep for backward compatibility)
270 RCLCPP_INFO_STREAM(getLogger(), "[" << getName() << "] getting current state.. waiting");
271
272 auto currentState = movegroupClient_->moveGroupClientInterface->getCurrentState(100);
273 auto groupname = movegroupClient_->moveGroupClientInterface->getName();
274
275 RCLCPP_INFO_STREAM(getLogger(), "[" << getName() << "] getting joint names");
276 auto currentjointnames =
277 currentState->getJointModelGroup(groupname)->getActiveJointModelNames();
278
279 if (!tipLink_ || *tipLink_ == "")
280 {
281 tipLink_ = movegroupClient_->moveGroupClientInterface->getEndEffectorLink();
282 }
283
284 std::vector<double> jointPositions;
285 currentState->copyJointGroupPositions(groupname, jointPositions);
286
287 std::vector<std::vector<double>> trajectory;
288 std::vector<rclcpp::Duration> trajectoryTimeStamps;
289
290 trajectory.push_back(jointPositions);
291 trajectoryTimeStamps.push_back(rclcpp::Duration(0s));
292
293 auto & first = endEffectorTrajectory_.front();
294 rclcpp::Time referenceTime(first.header.stamp);
295
296 std::vector<int> discontinuityIndexes;
297
298 int ikAttempts = 4;
299 for (size_t k = 0; k < this->endEffectorTrajectory_.size(); k++)
300 {
301 auto & pose = this->endEffectorTrajectory_[k];
302 auto req = std::make_shared<moveit_msgs::srv::GetPositionIK::Request>();
303
304 req->ik_request.ik_link_name = *tipLink_;
305 req->ik_request.robot_state.joint_state.name = currentjointnames;
306 req->ik_request.robot_state.joint_state.position = jointPositions;
307 req->ik_request.group_name = groupname;
308 req->ik_request.avoid_collisions = true;
309 req->ik_request.pose_stamped = pose;
310
311 RCLCPP_INFO_STREAM(getLogger(), "[" << getName() << "] IK request: " << k << " " << *req);
312
313 auto resfut = iksrv_->async_send_request(req);
314 auto status = resfut.wait_for(3s);
315
316 if (status == std::future_status::ready)
317 {
318 auto & prevtrajpoint = trajectory.back();
319 auto res = resfut.get();
320
321 std::stringstream ss;
322 for (size_t j = 0; j < res->solution.joint_state.position.size(); j++)
323 {
324 auto & jointname = res->solution.joint_state.name[j];
325 auto it = std::find(currentjointnames.begin(), currentjointnames.end(), jointname);
326 if (it != currentjointnames.end())
327 {
328 int index = std::distance(currentjointnames.begin(), it);
329 jointPositions[index] = res->solution.joint_state.position[j];
330 ss << jointname << "(" << index << "): " << jointPositions[index] << std::endl;
331 }
332 }
333
334 // Continuity check
335 size_t jointindex = 0;
336 int discontinuityJointIndex = -1;
337 double discontinuityDeltaJointIndex = -1;
338 double deltajoint;
339
340 bool check = k > 0 || !allowInitialTrajectoryStateJointDiscontinuity_ ||
341 (allowInitialTrajectoryStateJointDiscontinuity_ &&
342 !(*allowInitialTrajectoryStateJointDiscontinuity_));
343 if (check)
344 {
345 for (jointindex = 0; jointindex < jointPositions.size(); jointindex++)
346 {
347 deltajoint = jointPositions[jointindex] - prevtrajpoint[jointindex];
348 if (fabs(deltajoint) > 0.3)
349 {
350 discontinuityDeltaJointIndex = deltajoint;
351 discontinuityJointIndex = jointindex;
352 }
353 }
354 }
355
356 if (ikAttempts > 0 && discontinuityJointIndex != -1)
357 {
358 k--;
359 ikAttempts--;
360 continue;
361 }
362 else
363 {
364 bool discontinuity = false;
365 if (ikAttempts == 0)
366 {
367 discontinuityIndexes.push_back(k);
368 discontinuity = true;
369 }
370
371 ikAttempts = 4;
372
373 if (discontinuity && discontinuityJointIndex != -1)
374 {
375 std::stringstream ss;
376 ss << "Traj[" << k << "/" << endEffectorTrajectory_.size() << "] "
377 << currentjointnames[discontinuityJointIndex]
378 << " IK discontinuity : " << discontinuityDeltaJointIndex << std::endl
379 << "prev joint value: " << prevtrajpoint[discontinuityJointIndex] << std::endl
380 << "current joint value: " << jointPositions[discontinuityJointIndex] << std::endl;
381
382 ss << std::endl;
383 for (size_t ji = 0; ji < jointPositions.size(); ji++)
384 {
385 ss << currentjointnames[ji] << ": " << jointPositions[ji] << std::endl;
386 }
387
388 for (size_t kindex = 0; kindex < trajectory.size(); kindex++)
389 {
390 ss << "[" << kindex << "]: " << trajectory[kindex][discontinuityJointIndex]
391 << std::endl;
392 }
393
394 if (k == 0)
395 {
396 ss
397 << "This is the first posture of the trajectory. Maybe the robot initial posture "
398 "is "
399 "not coincident to the initial posture of the generated joint trajectory."
400 << std::endl;
401 }
402
403 RCLCPP_ERROR_STREAM(getLogger(), ss.str());
404
405 trajectory.push_back(jointPositions);
406 rclcpp::Duration durationFromStart = rclcpp::Time(pose.header.stamp) - referenceTime;
407 trajectoryTimeStamps.push_back(durationFromStart);
408 continue;
409 }
410 else
411 {
412 trajectory.push_back(jointPositions);
413 rclcpp::Duration durationFromStart = rclcpp::Time(pose.header.stamp) - referenceTime;
414 trajectoryTimeStamps.push_back(durationFromStart);
415
416 RCLCPP_DEBUG_STREAM(getLogger(), "IK solution: " << res->solution.joint_state);
417 RCLCPP_DEBUG_STREAM(getLogger(), "trajpoint: " << std::endl << ss.str());
418 }
419 }
420 }
421 else
422 {
423 RCLCPP_ERROR_STREAM(getLogger(), "[" << getName() << "] wrong IK call");
424 }
425 }
426
427 computedJointTrajectory.joint_trajectory.joint_names = currentjointnames;
428 int i = 0;
429 for (auto & p : trajectory)
430 {
431 if (i == 0) // Skip current state
432 {
433 i++;
434 continue;
435 }
436
437 trajectory_msgs::msg::JointTrajectoryPoint jp;
438 jp.positions = p;
439 jp.time_from_start = trajectoryTimeStamps[i];
440 computedJointTrajectory.joint_trajectory.points.push_back(jp);
441 i++;
442 }
443
444 if (discontinuityIndexes.size())
445 {
446 if (discontinuityIndexes[0] == 0)
447 return ComputeJointTrajectoryErrorCode::INCORRECT_INITIAL_STATE;
448 else
449 return ComputeJointTrajectoryErrorCode::JOINT_TRAJECTORY_DISCONTINUITY;
450 }
451
452 return ComputeJointTrajectoryErrorCode::SUCCESS;
453 }
454 }

456 void executeJointSpaceTrajectory(
457 const moveit_msgs::msg::RobotTrajectory & computedJointTrajectory)
458 {
459 RCLCPP_INFO_STREAM(getLogger(), "[" << this->getName() << "] Executing joint trajectory");
460
461 // Try to use CpTrajectoryExecutor component (preferred)
462 CpTrajectoryExecutor * trajectoryExecutor = nullptr;
463 this->requiresComponent(trajectoryExecutor, false); // Optional component
464
465 bool executionSuccess = false;
466
467 if (trajectoryExecutor != nullptr)
468 {
469 // Use component-based trajectory executor (preferred)
470 RCLCPP_INFO(
471 getLogger(),
472 "[CbMoveEndEffectorTrajectory] Using CpTrajectoryExecutor component for execution");
473
474 ExecutionOptions execOptions;
475 execOptions.trajectoryName = this->getName();
476
477 auto execResult = trajectoryExecutor->execute(computedJointTrajectory, execOptions);
478 executionSuccess = execResult.success;
479
480 if (executionSuccess)
481 {
482 RCLCPP_INFO(
483 getLogger(),
484 "[CbMoveEndEffectorTrajectory] Execution succeeded (via CpTrajectoryExecutor)");
485 }
486 else
487 {
488 RCLCPP_WARN(
489 getLogger(),
490 "[CbMoveEndEffectorTrajectory] Execution failed (via CpTrajectoryExecutor): %s",
491 execResult.errorMessage.c_str());
492 }
493 }
494 else
495 {
496 // Fallback to legacy direct execution
497 RCLCPP_WARN(
498 getLogger(),
499 "[CbMoveEndEffectorTrajectory] CpTrajectoryExecutor component not available, using legacy "
500 "execution (consider adding CpTrajectoryExecutor component)");
501
502 auto executionResult =
503 this->movegroupClient_->moveGroupClientInterface->execute(computedJointTrajectory);
504 executionSuccess = (executionResult == moveit_msgs::msg::MoveItErrorCodes::SUCCESS);
505
506 RCLCPP_INFO(
507 getLogger(), "[CbMoveEndEffectorTrajectory] Execution %s (legacy mode)",
508 executionSuccess ? "succeeded" : "failed");
509 }
510
511 // Post events
512 if (executionSuccess)
513 {
514 RCLCPP_INFO_STREAM(getLogger(), "[" << this->getName() << "] motion execution succeeded");
515 movegroupClient_->postEventMotionExecutionSucceded();
516 this->postSuccessEvent();
517 }
518 else
519 {
520 this->postMotionExecutionFailureEvents();
521 this->postFailureEvent();
522 }
523 }

545 marker.type = visualization_msgs::msg::Marker::ARROW;

552 marker.color.g = 0;
553 marker.color.b = 0;
554

556 start.x = 0;
557 start.y = 0;
558 start.z = 0;
559
560 tf2::Transform basetransform;

569 marker.pose.orientation = pose.pose.orientation;
570 marker.points.push_back(start);
571 marker.points.push_back(end);

583 void getCurrentEndEffectorPose(
584 std::string globalFrame, tf2::Stamped<tf2::Transform> & currentEndEffectorTransform)
585 {
586 // Use CpTfListener component for transform lookups
587 CpTfListener * tfListener = nullptr;
588 this->requiresComponent(tfListener, false); // Optional component
589
590 try
591 {
592 if (!tipLink_ || *tipLink_ == "")
593 {
594 tipLink_ = this->movegroupClient_->moveGroupClientInterface->getEndEffectorLink();
595 }
596
597 if (tfListener != nullptr)
598 {
599 // Use component-based TF listener (preferred)
600 auto transformOpt = tfListener->lookupTransform(globalFrame, *tipLink_, rclcpp::Time(0));
601 if (transformOpt)
602 {
603 tf2::fromMsg(transformOpt.value(), currentEndEffectorTransform);
604 }
605 else
606 {
607 RCLCPP_ERROR_STREAM(
608 getLogger(), "[" << getName() << "] Failed to lookup transform from " << *tipLink_
609 << " to " << globalFrame);
610 }
611 }
612 else
613 {
614 // Fallback to legacy TF2 usage if component not available
615 RCLCPP_WARN_STREAM(
616 getLogger(), "[" << getName()
617 << "] CpTfListener component not available, using legacy TF2 (consider "
618 "adding CpTfListener component)");
619 tf2_ros::Buffer tfBuffer(getNode()->get_clock());
620 tf2_ros::TransformListener tfListenerLegacy(tfBuffer);
621
622 tf2::fromMsg(
623 tfBuffer.lookupTransform(globalFrame, *tipLink_, rclcpp::Time(0), rclcpp::Duration(10s)),
624 currentEndEffectorTransform);
625 }
626 }
627 catch (const std::exception & e)
628 {
629 RCLCPP_ERROR_STREAM(
630 getLogger(), "[" << getName() << "] Exception in getCurrentEndEffectorPose: " << e.what());
