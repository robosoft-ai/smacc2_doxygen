23// Figure eight (lemniscate of Bernoulli) about a point (default: the current
24// position) on the path follower: entered at the nearest point of the curve,
25// flown `count` times at the follower's ground speed, yaw along the path so a
26// forward camera films where the vehicle is going. Unlike CbFigureEight (which
27// streams setpoints with a fixed heading) it is leashed and terminates on
28// arrival at the last vertex.
29class CbLemniscate : public CbPx4PathFollowerBase
30{
31public:
32 explicit CbLemniscate(
33 FlightPatternFigureEightParams params = {}, PathFollowerParams follower = {});
34 explicit CbLemniscate(int count, float size = 5.0f);
35
36 void setParams(const FlightPatternFigureEightParams & params) { params_ = params; }
37 const FlightPatternFigureEightParams & params() const { return params_; }
38 void setCenter(float x, float y)
39 {
40 params_.centerX = x;
41 params_.centerY = y;
42 }
43
44protected:
45 std::vector<NedPoint> buildPath(const NedPoint & current) override;
46 const char * behaviorName() const override { return "CbLemniscate"; }
47
48private:
49 FlightPatternFigureEightParams params_;
50};
