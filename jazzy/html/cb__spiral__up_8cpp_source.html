24void CbSpiralUp::onEntry()
25{
26 const float dx = localPosition_->getX() - params_.centerX;
27 const float dy = localPosition_->getY() - params_.centerY;
28 startAngle_ = std::atan2(dy, dx);
29 currentAngle_ = startAngle_;
30 startAltitude_ =
31 std::isfinite(params_.startAltitudeM) ? params_.startAltitudeM : -localPosition_->getZ();
32 lastReportedOrbit_ = -1;
33 lastUpdateTime_ = std::chrono::steady_clock::now();
34
35 RCLCPP_INFO(
36 getLogger(),
37 "CbSpiralUp: helix about NED (%.1f, %.1f), radius %.1f m, from %.1f m up %.1f m at %.1f m per "
38 "orbit (%.1f orbits), %.2f rad/s",
39 static_cast<double>(params_.centerX), static_cast<double>(params_.centerY),
40 static_cast<double>(params_.radiusM), static_cast<double>(startAltitude_),
41 static_cast<double>(params_.climbTotalM), static_cast<double>(params_.climbPerOrbitM),
42 static_cast<double>(params_.climbTotalM / params_.climbPerOrbitM),
43 static_cast<double>(params_.angularVelocityRadS));
44 command();
45}

47void CbSpiralUp::command()
48{
49 const float turned = currentAngle_ - startAngle_;
50 const float orbits = std::fabs(turned) / (2.0f * static_cast<float>(M_PI));
51 const float climb = std::min(orbits * params_.climbPerOrbitM, params_.climbTotalM);
52 const float x = params_.centerX + params_.radiusM * std::cos(currentAngle_);
53 const float y = params_.centerY + params_.radiusM * std::sin(currentAngle_);
54 const float z = -(startAltitude_ + climb); // NED: up is negative
55 const float yaw = currentAngle_ + static_cast<float>(M_PI); // nose toward the axis
56 trajectorySetpoint_->setPositionNED(x, y, z, yaw);
57
58 const int orbit = static_cast<int>(orbits);
59 if (orbit != lastReportedOrbit_)
60 {
61 lastReportedOrbit_ = orbit;
62 RCLCPP_INFO(
63 getLogger(), "CbSpiralUp: orbit %d, commanding %.1f m up", orbit,
64 static_cast<double>(startAltitude_ + climb));
65 }
66}

68void CbSpiralUp::update()
69{
70 CbPx4ClientBehaviorBase::update();
71
72 const auto now = std::chrono::steady_clock::now();
73 const double dt = std::chrono::duration<double>(now - lastUpdateTime_).count();
74 lastUpdateTime_ = now;
75 currentAngle_ += params_.angularVelocityRadS * static_cast<float>(dt);
76 command();
77
78 const float turned = std::fabs(currentAngle_ - startAngle_);
79 const float needed =
80 params_.climbTotalM / params_.climbPerOrbitM * 2.0f * static_cast<float>(M_PI);
81 if (turned >= needed)
82 {
83 RCLCPP_INFO(
84 getLogger(), "CbSpiralUp: %.1f m gained over %.1f orbits - posting success",
85 static_cast<double>(params_.climbTotalM), static_cast<double>(turned / (2.0f * M_PI)));
86 this->postPx4Success();
87 }
88}
