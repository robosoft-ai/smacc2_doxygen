35void CpRos2Timer::onInitialize()
36{
37 if (!initialized_)
38 {
39 RCLCPP_INFO_STREAM(
40 getLogger(), "[" << this->getName() << "] Initializing ROS2 Timer with duration: "
41 << duration_.seconds() << "s, oneshot: " << (oneshot_ ? "true" : "false"));
42
43 auto clock = this->getNode()->get_clock();
44 timer_ = rclcpp::create_timer(
45 this->getNode(), clock, duration_, std::bind(&CpRos2Timer::timerCallback, this));
46
47 this->initialized_ = true;
48 }
49}

51void CpRos2Timer::timerCallback()
52{
53 RCLCPP_DEBUG_STREAM(getLogger(), "[" << this->getName() << "] Timer tick");
54
55 if (!onTimerTick_.empty())
56 {
57 onTimerTick_();
58 }
59
60 if (oneshot_)
61 {
62 RCLCPP_INFO_STREAM(
63 getLogger(), "[" << this->getName() << "] Oneshot timer completed, cancelling");
64 timer_->cancel();
65 }
66}
