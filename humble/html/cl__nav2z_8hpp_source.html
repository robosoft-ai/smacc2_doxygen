47 // Component composition during orthogonal initialization
48 template <typename TOrthogonal, typename TClient>
49 void onComponentInitialization()
50 {
51 // Create core action client component

53 smacc2::client_core_components::CpActionClient<ActionType>, TOrthogonal, ClNav2Z>();
54 actionClient->actionServerName = actionServerName_;

56 // Create nav2-specific interface component (requires actionClient)
